Transaction Management Overview

Spring Boot provides powerful transaction management capabilities that abstract away the complexity of dealing with different transaction APIs (JTA, JPA, JDBC). The framework ensures data integrity and consistency through declarative and programmatic transaction management.

ACID Properties: Spring transactions ensure Atomicity, Consistency, Isolation, and Durability for database operations, maintaining data integrity across complex operations.

Basic @Transactional Usage


import org.springframework.transaction.annotation.Transactional;
import org.springframework.stereotype.Service;
import org.springframework.beans.factory.annotation.Autowired;

@Service
public class BankService {
    
    @Autowired
    private AccountRepository accountRepository;
    
    @Autowired
    private TransactionRepository transactionRepository;
    
    // Basic transaction
    @Transactional
    public void transferMoney(Long fromAccountId, Long toAccountId, BigDecimal amount) {
        // All database operations in this method are atomic
        Account fromAccount = accountRepository.findById(fromAccountId)
            .orElseThrow(() -> new AccountNotFoundException("From account not found"));
        
        Account toAccount = accountRepository.findById(toAccountId)
            .orElseThrow(() -> new AccountNotFoundException("To account not found"));
        
        // Check sufficient balance
        if (fromAccount.getBalance().compareTo(amount) < 0) {
            throw new InsufficientFundsException("Insufficient balance");
        }
        
        // Perform transfer
        fromAccount.debit(amount);
        toAccount.credit(amount);
        
        // Update accounts
        accountRepository.save(fromAccount);
        accountRepository.save(toAccount);
        
        // Create transaction record
        TransactionRecord record = new TransactionRecord(
            fromAccountId, 
            toAccountId, 
            amount, 
            LocalDateTime.now()
        );
        transactionRepository.save(record);
        
        // If any exception occurs here, ALL operations are rolled back
    }
    
    // Read-only transaction
    @Transactional(readOnly = true)
    public BigDecimal getAccountBalance(Long accountId) {
        // Optimized for read operations
        return accountRepository.findBalanceById(accountId);
    }
    
    // Transaction with specific rollback rules
    @Transactional(rollbackFor = {InsufficientFundsException.class, AccountNotFoundException.class})
    public void processBatchTransfer(List<TransferRequest> transfers) {
        for (TransferRequest transfer : transfers) {
            transferMoney(transfer.getFromAccountId(), 
                         transfer.getToAccountId(), 
                         transfer.getAmount());
        }
    }
    
    // No rollback for specific exception
    @Transactional(noRollbackFor = {NotificationFailedException.class})
    public void transferWithNotification(Long fromAccountId, Long toAccountId, BigDecimal amount) {
        transferMoney(fromAccountId, toAccountId, amount);
        
        try {
            notificationService.sendTransferNotification(fromAccountId, toAccountId, amount);
        } catch (NotificationFailedException e) {
            // Log but don't rollback the money transfer
            logger.error("Notification failed, but transfer completed", e);
        }
    }
}

2. Transaction Propagation

Propagation Types Explained

REQUIRED (Default)

Support current transaction, create a new one if none exists.

@Transactional(propagation = Propagation.REQUIRED)
Use Case: Most common scenario
REQUIRES_NEW

Create a new transaction, suspend the current one if exists.

@Transactional(propagation = Propagation.REQUIRES_NEW)
Use Case: Independent operations, audit logging
NESTED

Execute within nested transaction, uses savepoints.

@Transactional(propagation = Propagation.NESTED)
Use Case: Partial rollback scenarios
SUPPORTS

Support current transaction, execute non-transactionally if none.

@Transactional(propagation = Propagation.SUPPORTS)
Use Case: Read operations that can work with or without transaction 

@Service
public class PropagationExamplesService {
    
    @Autowired
    private AccountRepository accountRepository;
    
    @Autowired
    private AuditService auditService;
    
    @Autowired
    private NotificationService notificationService;
    
    // REQUIRED (Default behavior)
    @Transactional(propagation = Propagation.REQUIRED)
    public void requiredExample(Long accountId, BigDecimal amount) {
        // Uses existing transaction or creates new one
        Account account = accountRepository.findById(accountId).orElseThrow();
        account.deposit(amount);
        accountRepository.save(account);
    }
    
    // REQUIRES_NEW - Independent transaction
    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public void auditLog(String action, Long accountId) {
        // This runs in its own transaction
        // Commits even if calling method fails
        AuditLog log = new AuditLog(action, accountId, LocalDateTime.now());
        auditRepository.save(log);
    }
    
    // NESTED - Partial rollback capability
    @Transactional(propagation = Propagation.NESTED)
    public void processOrderItems(Order order) {
        // Uses savepoint - can rollback only this part
        for (OrderItem item : order.getItems()) {
            processItem(item);
        }
    }
    
    // MANDATORY - Must be called within transaction
    @Transactional(propagation = Propagation.MANDATORY)
    public void mandatoryTransactionMethod() {
        // Throws exception if no transaction exists
        // Use when method should only be called within transaction context
    }
    
    // NEVER - Must NOT be called within transaction
    @Transactional(propagation = Propagation.NEVER)
    public void sendExternalNotification(Notification notification) {
        // Throws exception if called within transaction
        // Use for external calls that shouldn't be in transaction
        externalNotificationService.send(notification);
    }
    
    // SUPPORTS - Works with or without transaction
    @Transactional(propagation = Propagation.SUPPORTS, readOnly = true)
    public Account getAccountDetails(Long accountId) {
        // Can be called with or without existing transaction
        return accountRepository.findById(accountId).orElseThrow();
    }
    
    // NOT_SUPPORTED - Suspend current transaction if exists
    @Transactional(propagation = Propagation.NOT_SUPPORTED)
    public void generateReport(Long accountId) {
        // Executes non-transactionally
        // Use for long-running operations that don't need transaction
        Report report = reportService.generateMonthlyReport(accountId);
        reportRepository.save(report);
    }
    
    // Real-world example combining propagations
    @Transactional
    public void completeBankTransfer(TransferRequest request) {
        try {
            // Main transaction
            transferMoney(request.getFromAccountId(), 
                         request.getToAccountId(), 
                         request.getAmount());
            
            // REQUIRES_NEW - Independent audit log
            auditService.logTransfer(request);
            
            // NESTED - Process fees (can rollback independently)
            processTransferFees(request);
            
            // NOT_SUPPORTED - Send notification (non-transactional)
            notificationService.sendTransferNotification(request);
            
        } catch (Exception e) {
            // REQUIRES_NEW - Log failure independently
            auditService.logTransferFailure(request, e);
            throw e;
        }
    }
}

3. Transaction Isolation Levels

Isolation Levels & Concurrency Control

Concurrency Problems
  • Dirty Read: Reading uncommitted data
  • Non-repeatable Read: Different reads yield different results
  • Phantom Read: New rows appear between reads
  • Lost Update: Overwritten concurrent updates
Isolation vs Performance

Higher isolation = Better consistency but Lower performance
Lower isolation = Better performance but Consistency risks

Isolation Level Dirty Read Non-repeatable Read Phantom Read Performance Use Case
READ_UNCOMMITTED YES YES YES HIGH Reporting, analytics
READ_COMMITTED NO YES YES HIGH Default for most databases
REPEATABLE_READ NO NO YES MEDIUM Financial transactions
SERIALIZABLE NO NO NO LOW Critical operations 

@Service
public class IsolationExamplesService {
    
    @Autowired
    private BankAccountRepository bankAccountRepository;
    
    @Autowired
    private TransactionHistoryRepository historyRepository;
    
    // READ_COMMITTED (Default for most databases)
    @Transactional(isolation = Isolation.READ_COMMITTED)
    public void processPayment(Long accountId, BigDecimal amount) {
        // Prevents dirty reads, allows non-repeatable and phantom reads
        BankAccount account = bankAccountRepository.findById(accountId).orElseThrow();
        
        // Other transactions can modify this account after we read it
        // But they can't read our uncommitted changes
        
        account.withdraw(amount);
        bankAccountRepository.save(account);
    }
    
    // REPEATABLE_READ - Prevents non-repeatable reads
    @Transactional(isolation = Isolation.REPEATABLE_READ)
    public BigDecimal calculateInterest(Long accountId) {
        // Multiple reads of same record will return same data
        BankAccount account = bankAccountRepository.findById(accountId).orElseThrow();
        
        // First read
        BigDecimal balance = account.getBalance();
        
        // Some processing time
        processComplexCalculation();
        
        // Second read - guaranteed to be same as first (for same records)
        // Other transactions can insert new rows (phantom reads possible)
        BigDecimal currentBalance = bankAccountRepository.findById(accountId)
            .orElseThrow()
            .getBalance();
        
        if (!balance.equals(currentBalance)) {
            // This should not happen with REPEATABLE_READ for the same record
            throw new IllegalStateException("Non-repeatable read detected!");
        }
        
        return calculateInterest(balance);
    }
    
    // SERIALIZABLE - Highest isolation, prevents all anomalies
    @Transactional(isolation = Isolation.SERIALIZABLE)
    public void transferWithSerializable(Long fromId, Long toId, BigDecimal amount) {
        // Locks entire table/range, prevents all concurrency issues
        // Use only for critical operations
        
        BankAccount fromAccount = bankAccountRepository.findById(fromId).orElseThrow();
        BankAccount toAccount = bankAccountRepository.findById(toId).orElseThrow();
        
        // These operations are completely isolated
        fromAccount.withdraw(amount);
        toAccount.deposit(amount);
        
        bankAccountRepository.save(fromAccount);
        bankAccountRepository.save(toAccount);
        
        // Performance impact: Other transactions wait for this to complete
    }
    
    // READ_UNCOMMITTED - Lowest isolation, for reporting
    @Transactional(isolation = Isolation.READ_UNCOMMITTED, readOnly = true)
    public Report generateRealTimeReport() {
        // Can read uncommitted data from other transactions
        // Use only for reporting where exact consistency isn't critical
        
        List<Transaction> recentTransactions = historyRepository
            .findRecentUncommittedTransactions();
        
        // WARNING: May include transactions that will be rolled back!
        return new Report(recentTransactions);
    }
    
    // Database-specific default isolation
    @Transactional  // Uses database default (usually READ_COMMITTED)
    public void defaultIsolationOperation() {
        // Good for most operations
        // Check your database default
    }
    
    // Mixed isolation levels in same service
    @Transactional(isolation = Isolation.REPEATABLE_READ)
    public void processFinancialBatch(List<FinancialOperation> operations) {
        for (FinancialOperation op : operations) {
            // High isolation for financial operations
            processFinancialOperation(op);
            
            // Lower isolation for logging
            logOperationWithReadCommitted(op);
        }
    }
    
    @Transactional(isolation = Isolation.READ_COMMITTED, 
                   propagation = Propagation.REQUIRES_NEW)
    public void logOperationWithReadCommitted(FinancialOperation op) {
        // Separate transaction with lower isolation for performance
        AuditLog log = new AuditLog("FINANCIAL_OP", op.getId());
        auditRepository.save(log);
    }
    
    // Handling isolation level conflicts
    @Transactional(isolation = Isolation.SERIALIZABLE, timeout = 10)
    public void criticalOperationWithTimeout() {
        try {
            // Critical operation requiring highest isolation
            performCriticalUpdate();
            
        } catch (CannotSerializeTransactionException e) {
            // Handle serialization failure (deadlock, etc.)
            logger.warn("Serializable transaction failed, retrying with lower isolation");
            
            // Fallback to lower isolation
            retryWithRepeatableRead();
        }
    }
    
    @Transactional(isolation = Isolation.REPEATABLE_READ)
    public void retryWithRepeatableRead() {
        // Fallback implementation
        performCriticalUpdate();
    }
}

public class ProgrammaticTransactionService {

    public void transferMoneyProgrammatically(Long fromId, Long toId, BigDecimal amount) {
        transactionTemplate.execute(status -> {

            Account toAccount = accountRepository.findById(toId).orElseThrow();
            
            fromAccount.withdraw(amount);
            toAccount.deposit(amount);

        });
    }
    
    // With return value
    public BigDecimal getBalanceWithTransaction(Long accountId) {
        return transactionTemplate.execute(status -> {
            Account account = accountRepository.findById(accountId).orElseThrow();
            return account.getBalance();
        });
    }
    
    // Multiple transactions in one method
    public void complexOperationWithMultipleTransactions(OperationRequest request) {
        // Transaction 1: Validation and setup
        ValidationResult validation = transactionTemplate.execute(status -> {
            return validateOperation(request);
        });
        
        if (!validation.isValid()) {
            throw new ValidationException(validation.getErrors());
        }
        
        // Transaction 2: Main operation (REQUIRES_NEW)
        OperationResult result = requiresNewTransactionTemplate.execute(status -> {
            return performMainOperation(request);
        });
        
        // Transaction 3: Audit logging (independent)
        requiresNewTransactionTemplate.execute(status -> {
            auditRepository.save(new AuditLog("OPERATION_COMPLETED", request.getId()));
            return null;
        });
        
        // Transaction 4: Cleanup (if needed)
        if (result.needsCleanup()) {
            transactionTemplate.execute(status -> {
                cleanupOperation(result);
                return null;
            });
        }
    }
    
    // Transaction with custom rollback rules
    public void operationWithCustomRollback(Data data) {
        transactionTemplate.execute(new TransactionCallbackWithoutResult() {
            @Override
            protected void doInTransactionWithoutResult(TransactionStatus status) {
                try {
                    processData(data);
                    
                    if (shouldRollback(data)) {
                        status.setRollbackOnly(); // Mark for rollback
                    }
                    
                } catch (BusinessException e) {
                    status.setRollbackOnly();
                    throw e;
                }
            }
        });
    }
    
    // Manual savepoint management
    public void operationWithSavepoints(ComplexData data) {
        transactionTemplate.execute(status -> {
            // Main operation
            processPart1(data);
            
            // Create savepoint
            Object savepoint = status.createSavepoint();
            
            try {
                // Risky operation that might fail
                processRiskyPart(data);
                
            } catch (RiskyOperationException e) {
                // Rollback only the risky part
                status.rollbackToSavepoint(savepoint);
                logger.warn("Risky operation failed, continuing with main", e);
                
                // Continue with alternative
                processAlternative(data);
            }
            
            // Continue with main operation
            processPart3(data);
            
            return null;
        });
    }
    
    // Chained transactions
    public void chainedTransactions(List<Operation> operations) {
        for (Operation operation : operations) {
            boolean success = transactionTemplate.execute(status -> {
                try {
                    processOperation(operation);
                    return true;
                } catch (OperationFailedException e) {
                    // Mark for rollback of this transaction only
                    status.setRollbackOnly();
                    logger.error("Operation failed: {}", operation.getId(), e);
                    return false;
                }
            });
            
            if (!success) {
                // Continue with next operation despite failure
                logger.warn("Continuing after failed operation: {}", operation.getId());
            }
        }
    }
    
    // Transaction with timeout control
    public void operationWithDynamicTimeout(Data data, int timeoutSeconds) {
        TransactionTemplate dynamicTemplate = new TransactionTemplate(
            transactionTemplate.getTransactionManager());
        dynamicTemplate.setTimeout(timeoutSeconds);
        
        dynamicTemplate.execute(status -> {
            processWithTimeout(data, timeoutSeconds);
            return null;
        });
    }
    
    // Read-only operations
    public Report generateReportProgrammatically(Long accountId) {
        return readOnlyTransactionTemplate.execute(status -> {
            // Optimized for reading
            Account account = accountRepository.findById(accountId).orElseThrow();
            List<Transaction> transactions = transactionRepository
                .findByAccountId(accountId);
            
            return new Report(account, transactions);
        });
    }
}

// Using PlatformTransactionManager directly
@Service
public class PlatformTransactionManagerService {
    
    @Autowired
    private PlatformTransactionManager transactionManager;
    
    @Autowired
    private AccountRepository accountRepository;
    
    public void transferWithDirectManager(Long fromId, Long toId, BigDecimal amount) {
        // Define transaction attributes
        DefaultTransactionDefinition def = new DefaultTransactionDefinition();
        def.setIsolationLevel(TransactionDefinition.ISOLATION_READ_COMMITTED);
        def.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRED);
        def.setTimeout(30);
        
        // Start transaction
        TransactionStatus status = transactionManager.getTransaction(def);
        
        try {
            // Business logic
            Account fromAccount = accountRepository.findById(fromId).orElseThrow();
            Account toAccount = accountRepository.findById(toId).orElseThrow();
            
            fromAccount.withdraw(amount);
            toAccount.deposit(amount);
            
            accountRepository.save(fromAccount);
            accountRepository.save(toAccount);
            
            // Commit transaction
            transactionManager.commit(status);
            
        } catch (Exception e) {
            // Rollback on exception
            transactionManager.rollback(status);
            throw e;
        }
    }
    
    // Nested transactions with savepoints
    public void nestedOperationsWithSavepoints() {
        DefaultTransactionDefinition def = new DefaultTransactionDefinition();
        TransactionStatus status = transactionManager.getTransaction(def);
        
        try {
            // Operation 1
            performOperation1();
            
            // Create savepoint for operation 2
            Object savepoint = status.createSavepoint();
            
            try {
                // Operation 2 (can rollback independently)
                performOperation2();
            } catch (Operation2Exception e) {
                // Rollback only operation 2
                status.rollbackToSavepoint(savepoint);
                logger.warn("Operation 2 failed, continuing", e);
            }
            
            // Operation 3
            performOperation3();
            
            transactionManager.commit(status);
            
        } catch (Exception e) {
            transactionManager.rollback(status);
            throw e;
        }
    }

            depositWithCompensation(request);

            saga.complete();
            sagaRepository.save(saga);
            
        } catch (Exception e) {
            // Compensate completed steps
            compensateSaga(saga);
            throw e;
        }
    }
    
    private void withdrawWithCompensation(SagaTransfer request) {
        try {
            Account account = primaryAccountRepository

                .orElseThrow();

            primaryAccountRepository.save(account);

                "CREDIT_ACCOUNT",
                request.getFromAccountId(),

                // Log but continue with other compensations
            }
        }
        
        saga.fail();
        sagaRepository.save(saga);
    }

        try {
            // Attempt to complete or rollback
            if (canComplete(tx)) {
                completeTransaction(tx);
                tx.setStatus(PendingTransactionStatus.COMPLETED);
            } else {
                rollbackTransaction(tx);
                tx.setStatus(PendingTransactionStatus.ROLLED_BACK);
            }
            
            pendingTransactionRepository.save(tx);
            
        } catch (Exception e) {
            logger.error("Recovery failed for transaction: {}", tx.getId(), e);
            tx.setStatus(PendingTransactionStatus.RECOVERY_FAILED);
            pendingTransactionRepository.save(tx);
        }
    }

public class TransactionInterceptor extends TransactionAspectSupport 
    implements MethodInterceptor {
    
    @Override
    public Object invoke(MethodInvocation invocation) throws Throwable {
        // Get transaction attributes for the method
        TransactionAttribute txAttr = getTransactionAttributeSource()
            .getTransactionAttribute(invocation.getMethod(), 
                                     invocation.getThis().getClass());
        
        // Determine transaction manager
        PlatformTransactionManager tm = determineTransactionManager(txAttr);
        
        // Handle transaction based on propagation behavior
        if (txAttr == null || !(tm instanceof CallbackPreferringPlatformTransactionManager)) {
            // Standard transaction flow
            TransactionInfo txInfo = createTransactionIfNecessary(tm, txAttr, 
                invocation.getMethod().getQualifiedName());
            
            Object retVal;
            try {
                // Proceed with the original method
                retVal = invocation.proceed();
            } catch (Throwable ex) {
                // Handle rollback
                completeTransactionAfterThrowing(txInfo, ex);
                throw ex;
            } finally {
                cleanupTransactionInfo(txInfo);
            }
            
            // Commit transaction
            commitTransactionAfterReturning(txInfo);
            return retVal;
        }
        // ... additional handling for reactive/callback preferring managers
    }
}

// Internal transaction lifecycle in TransactionAspectSupport
public abstract class TransactionAspectSupport implements BeanFactoryAware {
    
    protected TransactionInfo createTransactionIfNecessary(
            PlatformTransactionManager tm, 
            TransactionAttribute txAttr, 
            final String joinpointIdentification) {
        
        // If no name specified, use method identification as transaction name
        if (txAttr != null && txAttr.getName() == null) {
            txAttr = new DelegatingTransactionAttribute(txAttr) {
                @Override
                public String getName() {
                    return joinpointIdentification;
                }
            };
        }
        
        TransactionStatus status = null;
        if (txAttr != null) {
            if (tm != null) {
                // Get transaction status (may create new or use existing)
                status = tm.getTransaction(txAttr);
            }
        }
        
        // Prepare TransactionInfo
        return prepareTransactionInfo(tm, txAttr, joinpointIdentification, status);
    }
    
    protected void commitTransactionAfterReturning(TransactionInfo txInfo) {
        if (txInfo != null && txInfo.hasTransaction()) {
            if (logger.isTraceEnabled()) {
                logger.trace("Completing transaction for [" + 
                    txInfo.getJoinpointIdentification() + "]");
            }
            txInfo.getTransactionManager().commit(txInfo.getTransactionStatus());
        }
    }
    
    protected void completeTransactionAfterThrowing(TransactionInfo txInfo, Throwable ex) {
        if (txInfo != null && txInfo.hasTransaction()) {
            if (logger.isTraceEnabled()) {
                logger.trace("Completing transaction for [" + 
                    txInfo.getJoinpointIdentification() + "] after exception: " + ex);
            }
            
            // Check rollback rules
            if (txInfo.transactionAttribute.rollbackOn(ex)) {
                try {
                    txInfo.getTransactionManager().rollback(txInfo.getTransactionStatus());
                } catch (TransactionSystemException ex2) {
                    logger.error("Application exception overridden by rollback exception", ex);
                    ex2.initApplicationException(ex);
                    throw ex2;
                } catch (RuntimeException | Error ex2) {
                    logger.error("Application exception overridden by rollback exception", ex);
                    throw ex2;
                }
            } else {
                // We don't roll back on this exception
                try {
                    txInfo.getTransactionManager().commit(txInfo.getTransactionStatus());
                } catch (TransactionSystemException ex2) {
                    logger.error("Application exception overridden by commit exception", ex);
                    ex2.initApplicationException(ex);
                    throw ex2;
                } catch (RuntimeException | Error ex2) {
                    logger.error("Application exception overridden by commit exception", ex);
                    throw ex2;
                }
            }
        }
    }
}

// Core PlatformTransactionManager interface
public interface PlatformTransactionManager extends TransactionManager {
    
    // Get transaction status (creates or participates in existing transaction)
    TransactionStatus getTransaction(TransactionDefinition definition) 
        throws TransactionException;
    
    // Commit the transaction
    void commit(TransactionStatus status) throws TransactionException;
    
    // Rollback the transaction
    void rollback(TransactionStatus status) throws TransactionException;
}

// Abstract base implementation
public abstract class AbstractPlatformTransactionManager 
    implements PlatformTransactionManager, Serializable {
    
    @Override
    public final TransactionStatus getTransaction(TransactionDefinition definition) 
        throws TransactionException {
        
        // Use defaults if no definition provided
        TransactionDefinition def = (definition != null ? definition : 
            TransactionDefinition.withDefaults());
        
        Object transaction = doGetTransaction();
        boolean debugEnabled = logger.isDebugEnabled();
        
        if (isExistingTransaction(transaction)) {
            // Existing transaction found → handle propagation behavior
            return handleExistingTransaction(def, transaction, debugEnabled);
        }
        
        // Check timeout
        if (def.getTimeout() < TransactionDefinition.TIMEOUT_DEFAULT) {
            throw new InvalidTimeoutException("Invalid transaction timeout", def.getTimeout());
        }
        
        // No existing transaction → create new one
        if (def.getPropagationBehavior() == TransactionDefinition.PROPAGATION_MANDATORY) {
            throw new IllegalTransactionStateException(
                "No existing transaction found for transaction marked with propagation 'mandatory'");
        } else if (def.getPropagationBehavior() == TransactionDefinition.PROPAGATION_REQUIRED ||
                   def.getPropagationBehavior() == TransactionDefinition.PROPAGATION_REQUIRES_NEW ||
                   def.getPropagationBehavior() == TransactionDefinition.PROPAGATION_NESTED) {
            
            SuspendedResourcesHolder suspendedResources = suspend(null);
            try {
                boolean newSynchronization = (getTransactionSynchronization() != SYNCHRONIZATION_NEVER);
                DefaultTransactionStatus status = newTransactionStatus(
                    def, transaction, true, newSynchronization, debugEnabled, suspendedResources);
                doBegin(transaction, def);
                prepareSynchronization(status, def);
                return status;
            } catch (RuntimeException | Error ex) {
                resume(null, suspendedResources);
                throw ex;
            }
        } else {
            // Create "empty" transaction: no actual transaction, but synchronization if required
            boolean newSynchronization = (getTransactionSynchronization() == SYNCHRONIZATION_ALWAYS);
            return prepareTransactionStatus(def, null, true, newSynchronization, debugEnabled, null);
        }
    }
    
    // Template methods to be implemented by subclasses
    protected abstract Object doGetTransaction() throws TransactionException;
    protected abstract boolean isExistingTransaction(Object transaction) throws TransactionException;
    protected abstract void doBegin(Object transaction, TransactionDefinition definition) 
        throws TransactionException;
    protected abstract void doCommit(DefaultTransactionStatus status) throws TransactionException;
    protected abstract void doRollback(DefaultTransactionStatus status) throws TransactionException;
    protected abstract void doSetRollbackOnly(DefaultTransactionStatus status) throws TransactionException;
}

// Concrete implementation: DataSourceTransactionManager (for JDBC)
public class DataSourceTransactionManager extends AbstractPlatformTransactionManager
    implements ResourceTransactionManager, InitializingBean {
    
    private DataSource dataSource;
    
    @Override
    protected Object doGetTransaction() {
        DataSourceTransactionObject txObject = new DataSourceTransactionObject();
        txObject.setSavepointAllowed(isNestedTransactionAllowed());
        
        ConnectionHolder conHolder = (ConnectionHolder) 
            TransactionSynchronizationManager.getResource(this.dataSource);
        txObject.setConnectionHolder(conHolder, false);
        return txObject;
    }
    
    @Override
    protected boolean isExistingTransaction(Object transaction) {
        DataSourceTransactionObject txObject = (DataSourceTransactionObject) transaction;
        return (txObject.getConnectionHolder() != null && 
                txObject.getConnectionHolder().isTransactionActive());
    }
    
    @Override
    protected void doBegin(Object transaction, TransactionDefinition definition) {
        DataSourceTransactionObject txObject = (DataSourceTransactionObject) transaction;
        Connection con = null;
        
        try {
            if (txObject.getConnectionHolder() == null || 
                txObject.getConnectionHolder().isSynchronizedWithTransaction()) {
                Connection newCon = this.dataSource.getConnection();
                txObject.setConnectionHolder(new ConnectionHolder(newCon), true);
            }
            
            txObject.getConnectionHolder().setSynchronizedWithTransaction(true);
            con = txObject.getConnectionHolder().getConnection();
            
            // Apply transaction isolation level
            Integer previousIsolationLevel = DataSourceUtils.prepareConnectionForTransaction(
                con, definition);
            txObject.setPreviousIsolationLevel(previousIsolationLevel);
            
            // Switch to manual commit if necessary
            if (con.getAutoCommit()) {
                txObject.setMustRestoreAutoCommit(true);
                con.setAutoCommit(false);
            }
            
            // Apply read-only flag
            txObject.getConnectionHolder().setReadOnly(definition.isReadOnly());
            
            // Apply transaction timeout
            if (definition.getTimeout() != TransactionDefinition.TIMEOUT_DEFAULT) {
                txObject.getConnectionHolder().setTimeoutInSeconds(definition.getTimeout());
            }
            
            // Bind the connection holder to the thread
            if (txObject.isNewConnectionHolder()) {
                TransactionSynchronizationManager.bindResource(
                    getDataSource(), txObject.getConnectionHolder());
            }
            
        } catch (Throwable ex) {
            DataSourceUtils.releaseConnection(con, this.dataSource);
            throw new CannotCreateTransactionException(
                "Could not open JDBC Connection for transaction", ex);
        }
    }
    
    @Override
    protected void doCommit(DefaultTransactionStatus status) {
        DataSourceTransactionObject txObject = (DataSourceTransactionObject) status.getTransaction();
        Connection con = txObject.getConnectionHolder().getConnection();
        
        try {
            con.commit();
        } catch (SQLException ex) {
            throw new TransactionSystemException("Could not commit JDBC transaction", ex);
        }
    }
    
    @Override
    protected void doRollback(DefaultTransactionStatus status) {
        DataSourceTransactionObject txObject = (DataSourceTransactionObject) status.getTransaction();
        Connection con = txObject.getConnectionHolder().getConnection();
        
        try {
            con.rollback();
        } catch (SQLException ex) {
            throw new TransactionSystemException("Could not roll back JDBC transaction", ex);
        }
    }
}

        return value;
    }

            pendingEvents.set(events);

            TransactionSynchronizationManager.registerSynchronization(this);

        events.add(event);

        // Validate all audit events before commit
        List<AuditEvent> events = pendingEvents.get();
        if (events != null) {
            for (AuditEvent event : events) {
                event.validate();
            }
        }
    }
    
    @Override
    public void afterCommit() {
        // Process audit events after successful commit
        List<AuditEvent> events = pendingEvents.get();
        if (events != null) {
            try {
                for (AuditEvent event : events) {
                    auditService.save(event);
                }
            } finally {
                pendingEvents.remove();
            }
        }
    }
    
    @Override
    public void afterCompletion(int status) {
        // Cleanup regardless of commit/rollback
        if (status != TransactionSynchronization.STATUS_COMMITTED) {
            pendingEvents.remove();
        }
    }

// AbstractPlatformTransactionManager.handleExistingTransaction()
protected TransactionStatus handleExistingTransaction(
        TransactionDefinition definition, Object transaction, boolean debugEnabled)
        throws TransactionException {
    
    // 1. NEVER propagation
    if (definition.getPropagationBehavior() == TransactionDefinition.PROPAGATION_NEVER) {
        throw new IllegalTransactionStateException(
            "Existing transaction found for transaction marked with propagation 'never'");
    }
    
    // 2. NOT_SUPPORTED propagation
    if (definition.getPropagationBehavior() == TransactionDefinition.PROPAGATION_NOT_SUPPORTED) {
        if (debugEnabled) {
            logger.debug("Suspending current transaction");
        }
        Object suspendedResources = suspend(transaction);
        boolean newSynchronization = (getTransactionSynchronization() == SYNCHRONIZATION_ALWAYS);
        return prepareTransactionStatus(
            definition, null, false, newSynchronization, debugEnabled, suspendedResources);
    }
    
    // 3. REQUIRES_NEW propagation
    if (definition.getPropagationBehavior() == TransactionDefinition.PROPAGATION_REQUIRES_NEW) {
        if (debugEnabled) {
            logger.debug("Suspending current transaction, creating new transaction with name [" +
                definition.getName() + "]");
        }
        SuspendedResourcesHolder suspendedResources = suspend(transaction);
        try {
            boolean newSynchronization = (getTransactionSynchronization() != SYNCHRONIZATION_NEVER);
            DefaultTransactionStatus status = newTransactionStatus(
                definition, transaction, true, newSynchronization, debugEnabled, suspendedResources);
            doBegin(transaction, definition);
            prepareSynchronization(status, definition);
            return status;
        } catch (RuntimeException | Error beginEx) {
            resumeAfterBeginException(transaction, suspendedResources, beginEx);
            throw beginEx;
        }
    }
    
    // 4. NESTED propagation
    if (definition.getPropagationBehavior() == TransactionDefinition.PROPAGATION_NESTED) {
        if (!isNestedTransactionAllowed()) {
            throw new NestedTransactionNotSupportedException(
                "Transaction manager does not allow nested transactions by default - " +
                "specify 'nestedTransactionAllowed' property with value 'true'");
        }
        if (debugEnabled) {
            logger.debug("Creating nested transaction with name [" + definition.getName() + "]");
        }
        if (useSavepointForNestedTransaction()) {
            // Create savepoint within existing Spring transaction
            DefaultTransactionStatus status = prepareTransactionStatus(
                definition, transaction, false, false, debugEnabled, null);
            status.createAndHoldSavepoint();
            return status;
        } else {
            // Nested transaction through nested begin and commit/rollback calls
            boolean newSynchronization = (getTransactionSynchronization() != SYNCHRONIZATION_NEVER);
            DefaultTransactionStatus status = newTransactionStatus(
                definition, transaction, true, newSynchronization, debugEnabled, null);
            doBegin(transaction, definition);
            prepareSynchronization(status, definition);

// How different API exceptions are translated
public abstract class AbstractPlatformTransactionManager {
    
    protected void commit(DefaultTransactionStatus status) {
        try {
            doCommit(status);  // Technology-specific commit
        } catch (RuntimeException ex) {
            // Translate technology-specific exceptions to Spring's unified exceptions
            throw translateCommitException(status, ex);
        }
    }
    
    private RuntimeException translateCommitException(
            DefaultTransactionStatus status, RuntimeException ex) {
        
        // Common translation logic for all technologies
        if (ex instanceof HeuristicCompletionException) {
            // JTA-specific → Spring common
            return new HeuristicCompletionException(
                "Heuristic completion: " + ex.getMessage(), ex);
        } else if (ex instanceof SQLException) {
            // JDBC-specific → Spring common
            return new TransactionSystemException(
                "JDBC commit failed: " + ex.getMessage(), ex);
        } else if (ex instanceof PersistenceException) {
            // JPA-specific → Spring common
            return new TransactionSystemException(
                "JPA commit failed: " + ex.getMessage(), ex);
        } else if (ex instanceof TransactionException) {
            // Already a Spring transaction exception
            return ex;
        } else {
            // Wrap any other exception
            return new TransactionSystemException(
                "Unexpected exception during commit", ex);
        }
    }
}

// Consistent rollback rules across technologies
public class RuleBasedTransactionAttribute extends DefaultTransactionAttribute {
    
    @Override
    public boolean rollbackOn(Throwable ex) {
        // Same rollback rules work for JDBC, JPA, JTA, etc.
        
        // Always rollback on RuntimeException and Error
        if (ex instanceof RuntimeException || ex instanceof Error) {
            return true;
        }
        
        // Check configured rollback rules
        if (this.rollbackRules != null) {
            for (RollbackRuleAttribute rule : this.rollbackRules) {
                if (rule.getDepth(ex) >= 0) {
                    return !rule.isNoRollbackRule();
                }
            }
        }
        
        // Default: don't rollback on checked exceptions
        return false;
    }
}

        } else if (ex instanceof TransactionSystemException) {
            return HttpStatus.INTERNAL_SERVER_ERROR;

            return HttpStatus.CONFLICT;
        }
    }

// Programmatic configuration unification
public abstract class TransactionConfigBuilder {
    
    // Common builder methods for all transaction types
    public TransactionConfigBuilder timeout(int seconds) {
        // Applies to JDBC, JPA, JTA
        this.timeout = seconds;
        return this;
    }
    
    public TransactionConfigBuilder isolation(Isolation isolation) {
        // Applies to JDBC, JPA, JTA (where supported)
        this.isolation = isolation;
        return this;
    }
    
    public TransactionConfigBuilder readOnly(boolean readOnly) {
        // Applies to JDBC, JPA, JTA
        this.readOnly = readOnly;
        return this;
    }
    
    // Factory method that returns appropriate manager
    public static PlatformTransactionManager createManager(
            TransactionConfig config, Object resource) {
        
        if (resource instanceof DataSource) {
            DataSourceTransactionManager tm = new DataSourceTransactionManager();
            tm.setDataSource((DataSource) resource);
            applyCommonConfig(tm, config);
            return tm;
            
        } else if (resource instanceof EntityManagerFactory) {
            JpaTransactionManager tm = new JpaTransactionManager();
            tm.setEntityManagerFactory((EntityManagerFactory) resource);
            applyCommonConfig(tm, config);
            return tm;
            
        } else if (resource instanceof UserTransaction) {
            JtaTransactionManager tm = new JtaTransactionManager();
            tm.setUserTransaction((UserTransaction) resource);
            applyCommonConfig(tm, config);
            return tm;
        }
        
        throw new IllegalArgumentException("Unsupported resource type: " + resource.getClass());
    }
    
    private static void applyCommonConfig(AbstractPlatformTransactionManager tm, 
                                          TransactionConfig config) {
        // Apply common configuration to ANY transaction manager
        tm.setDefaultTimeout(config.getTimeout());
        tm.setNestedTransactionAllowed(config.isNestedTransactionAllowed());
        tm.setValidateExistingTransaction(config.isValidateExistingTransaction());
        
        if (config.getIsolation() != null) {
            // Set isolation if manager supports it
            try {
                Method setIsolationMethod = tm.getClass()
                    .getMethod("setIsolationLevel", int.class);
                setIsolationMethod.invoke(tm, config.getIsolation().value());
            } catch (Exception e) {
                // Manager doesn't support isolation setting
                logger.debug("Transaction manager doesn't support isolation setting", e);
            }
        }
    }
}

Benefits of Transaction API Unification

Benefits

  • Consistent Programming Model: Same annotations work across technologies
  • Reduced Learning Curve: Learn once, apply everywhere
  • Technology Agnostic: Easy to switch between JDBC, JPA, JTA
  • Unified Testing: Same test code works with different transaction managers
  • Common Infrastructure: AOP, monitoring, logging work uniformly
  • Exception Consistency: Same exception handling for all technologies
  • Configuration Uniformity: Similar configuration patterns

Trade-offs & Limitations

  • Abstraction Overhead: Extra layer adds some complexity
  • Lowest Common Denominator: Only features available in all APIs are exposed
  • Performance Impact: Small overhead for abstraction layer
  • API-Specific Features: Some vendor-specific features may not be exposed
  • Configuration Complexity: Need to understand both Spring and underlying API
  • Debugging Complexity: Stack traces go through abstraction layer
  • Migration Challenges: Moving from Spring to raw API requires relearning

    @Autowired
    private PlatformTransactionManager transactionManager;
    
    public void executeWithIsolationControl(Operation operation, IsolationLevel level) {
        
        // Create isolation-specific transaction definition
        DefaultTransactionDefinition def = new DefaultTransactionDefinition();
        def.setName("IsolationControlledOperation");
        def.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRED);
        def.setIsolationLevel(level.getValue());
        def.setTimeout(30);
        
        // Execute
        TransactionStatus status = transactionManager.getTransaction(def);
        
        try {
            operation.execute();
            transactionManager.commit(status);
        } catch (Exception e) {
            transactionManager.rollback(status);
            throw e;
        }
    }
    
    // Switch isolation level mid-transaction (database dependent)
    public void switchIsolationLevel(Connection connection, IsolationLevel newLevel) 
            throws SQLException {
        
        int isolationValue;
        switch (newLevel) {
            case READ_UNCOMMITTED:
                isolationValue = Connection.TRANSACTION_READ_UNCOMMITTED;
                break;
            case READ_COMMITTED:
                isolationValue = Connection.TRANSACTION_READ_COMMITTED;
                break;
            case REPEATABLE_READ:
                isolationValue = Connection.TRANSACTION_REPEATABLE_READ;
                break;
            case SERIALIZABLE:
                isolationValue = Connection.TRANSACTION_SERIALIZABLE;
                break;
            default:
                isolationValue = Connection.TRANSACTION_READ_COMMITTED;
        }
        
        connection.setTransactionIsolation(isolationValue);
    }
}

Complete Transaction Management in Spring: Theory + Code

I. FUNDAMENTAL THEORY

1. What is a Transaction?

Theory: A logical unit of work that follows ACID properties (Atomicity, Consistency, Isolation, Durability). In Spring, transactions provide a way to execute multiple database operations as a single unit.

Code Example:

// WITHOUT Transactions (Problematic)
public void transferMoney(Long fromId, Long toId, BigDecimal amount) {
    // If this fails...
    accountRepository.debit(fromId, amount);
    // Money is debited but not credited!
    accountRepository.credit(toId, amount);
}

// WITH Transactions (Correct)
@Transactional
public void transferMoney(Long fromId, Long toId, BigDecimal amount) {
    // Both succeed or both fail together
    accountRepository.debit(fromId, amount);
    accountRepository.credit(toId, amount);
}

2. Spring's Transaction Abstraction Layer

Theory: Spring sits between your code and the underlying transaction API (JDBC, JTA, Hibernate, JPA), providing a consistent programming model.

III. PROPAGATION BEHAVIORS - THEORY & CODE

Theory: Defines how transactions relate to each other when methods call each other.

1. REQUIRED (Default)

Theory: Use existing transaction, create new if none exists.

@Service
public class AccountService {
    
    @Transactional(propagation = Propagation.REQUIRED)
    public void methodA() {
        // If called directly: Creates new transaction TX1
        // If called from another transactional method: Joins existing transaction

IV. ISOLATION LEVELS - THEORY & CODE

Theory: Controls visibility of changes between concurrent transactions.

1. READ_COMMITTED (Most Common)

Theory: Can only read committed data. Prevents dirty reads.

@Service
public class BankService {

    public BigDecimal calculateTotalBalance(Long customerId) {

        // UPDATE accounts SET balance = balance - 100 WHERE id = 1 (not committed yet)
        
        // Thread 2 (this method):
        BigDecimal total = accountRepository.sumBalances(customerId);
        // Won't see the uncommitted -100
        // Sees old balance until Thread 1 commits
        
        return total;
    }
}

Theory: Guarantees same data read multiple times in same transaction.

@Service

V. ROLLBACK RULES - THEORY & CODE

Theory: Defines which exceptions trigger rollback.

1. Default Behavior

@Service
public class DefaultRollbackService {
    
    @Transactional
    public void processData() {
        // DEFAULT: Rolls back on RuntimeException and Error
        throw new RuntimeException();      // ← ROLLBACK
        throw new NullPointerException();  // ← ROLLBACK
        throw new Error();                 // ← ROLLBACK
        
        // DEFAULT: Does NOT rollback on checked exceptions
        throw new Exception();             // ← COMMIT
        throw new SQLException();          // ← COMMIT
        throw new IOException();           // ← COMMIT
    }
}

2. Custom Rollback Rules

@Service
public class CustomRollbackService {
    
    // Rollback for specific exceptions
    @Transactional(rollbackFor = {SQLException.class, 
                                  IOException.class,
                                  BusinessException.class})
    public void importData() throws SQLException, BusinessException {
        try {
            dataImporter.importFromFile();  // Might throw IOException
            dataValidator.validate();        // Might throw BusinessException
            dataRepository.save();           // Might throw SQLException
        } catch (ValidationException e) {
            // This is a checked exception but NOT in rollbackFor
            // Transaction will COMMIT despite this exception!
            log.warn("Validation warning", e);
        }
    }
    
    // No rollback for specific exceptions
    @Transactional(noRollbackFor = {ValidationException.class,
                                    NotificationException.class})
    public void registerUser(User user) {
        userValidator.validate(user);  // Throws ValidationException (no rollback)
        userRepository.save(user);     // If this fails, rolls back
        
        try {
            emailService.sendWelcomeEmail(user);  // Throws NotificationException
        } catch (NotificationException e) {
            // Email failed, but user is saved (no rollback)
            log.error("Failed to send email", e);
        }
    }
    
    // Combined rules
    @Transactional(
        rollbackFor = {DataAccessException.class},
        noRollbackFor = {BusinessRuleException.class}
    )
    public void complexOperation() {
        // DataAccessException → Rollback
        // BusinessRuleException → Commit
        // Other RuntimeExceptions → Rollback (default)
        // Checked exceptions → Commit (default)
    }
}

@Service

            // Mark for rollback only
            TransactionAspectSupport.currentTransactionStatus()
                .setRollbackOnly();
            
            // Transaction will rollback even though no exception thrown
            log.warn("Order amount {} exceeds limit {}", 
                     order.getAmount(), MAX_LIMIT);
            return;
        }
        
        // Continue processing...
        inventoryService.reserve(order);
    }
    
    // Method 2: Using TransactionTemplate
    public void processWithTemplate(Data data) {
        transactionTemplate.execute(status -> {
            try {
                step1(data);
                step2(data);
                
                if (shouldRollback(data)) {
                    status.setRollbackOnly();
                    return "Rollback triggered";
                }
                
                step3(data);
                return "Success";
                
            } catch (BusinessException e) {
                // Handle exception but don't rollback
                log.error("Business error", e);
                return "Business error handled";
            }
        });
    }
    
    // Method 3: Declarative + Programmatic combination
    @Transactional
    public void mixedApproach(Invoice invoice) {
        invoiceRepository.save(invoice);
        
        try {
            externalService.submit(invoice);
        } catch (ExternalServiceException e) {
            // External service failed, but we want to keep invoice

            invoice.setStatus(InvoiceStatus.RETRY);
            invoiceRepository.save(invoice);
            
            // Don't rollback - we want to keep the invoice
            // Throw a checked exception (no rollback by default)
            throw new RetryableException("External service failed", e);
        }
    }

5. Distributed Transaction Pattern (Saga)

Theory: For microservices where traditional 2PC isn't feasible.

@Service
public class OrderSagaService {
    
    @Transactional
    public Order createOrderDistributed(OrderRequest request) {
        // Step 1: Create order locally
        Order order = createLocalOrder(request);
        
        try {
            // Step 2: Reserve inventory (remote service)
            inventoryService.reserve(order.getItems());
            
            // Step 3: Process payment (remote service)
            paymentService.process(order.getPayment());
            
            // Step 4: Confirm order
            order.confirm();
            orderRepository.save(order);
            
            return order;
            
        } catch (Exception e) {
            // Compensation: Rollback distributed transaction
            compensate(order, e);
            throw e;
        }
    }
    
    private void compensate(Order order, Exception cause) {
        log.error("Compensating order {} due to {}", order.getId(), cause.getMessage());
        
        try {
            // Reverse inventory reservation
            inventoryService.release(order.getItems());
        } catch (Exception ex) {
            log.error("Failed to release inventory for order {}", order.getId(), ex);
        }
        
        try {
            // Reverse payment if it was processed
            if (order.getPayment().isProcessed()) {
                paymentService.refund(order.getPayment());
            }
        } catch (Exception ex) {
            log.error("Failed to refund payment for order {}", order.getId(), ex);
        }
        
        // Mark order as failed
        order.fail(cause.getMessage());
        orderRepository.save(order);
    }
}

Transaction Propagation Scenarios & Implementation

Understanding Propagation Behaviors in Practice

1. How to create a new transaction for each method call?

Use Propagation.REQUIRES_NEW - Always creates a new transaction, suspending existing one if present.

Theory:
  • Independent transaction for each call
  • Suspends current transaction if exists
  • Commits/Rollbacks independently
  • Useful for audit logs, notifications
Use Case:

Audit logging that must persist even if main transaction fails.

@Service
public class AuditService {
    
    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public void logAction(String action, Long entityId) {
        // Always creates new transaction
        // Even if called from another @Transactional method
        auditRepository.save(new AuditLog(action, entityId, new Date()));
    }
}

@Service  
public class OrderService {
    
    @Transactional
    public Order processOrder(Order order) {
        // TX1 starts here
        
        // Save order in main transaction
        orderRepository.save(order);
        
        // TX1 suspended, TX2 created
        auditService.logAction("ORDER_CREATED", order.getId());
        // TX2 commits independently
        
        // TX1 resumes
        inventoryService.updateStock(order);
        // TX1 commits/rolls back
        return order;
    }
}

2. How to suspend current transaction and create new one?

Use Propagation.NOT_SUPPORTED - Executes non-transactionally, suspending any existing transaction.

Theory:
  • Suspends current transaction
  • Executes without any transaction
  • Resumes original transaction after completion
  • No commit/rollback performed
Use Case:

External API calls, file operations, or any non-database operations.

@Service
public class ExternalService {
    
    @Transactional(propagation = Propagation.NOT_SUPPORTED)
    public void callExternalAPI(Order order) {
        // If called within transaction, that transaction is suspended
        // No transaction active during this method execution
        
        // Call external REST API
        restTemplate.postForEntity("https://api.example.com/orders", order, String.class);
        
        // File operations
        Files.write(Paths.get("logs/order.log"), 
                   order.toString().getBytes());
        
        // Original transaction resumes after this method
    }
}

@Service
public class OrderService {
    
    @Transactional
    public Order createOrder(Order order) {
        // TX1 active
        
        orderRepository.save(order);  // In TX1
        
        // TX1 suspended, no transaction for API call
        externalService.callExternalAPI(order);
        
        // TX1 resumes
        inventoryService.reserve(order.getItems());  // Back in TX1
        
        return order;
    }
}

3. How to ensure method runs without any transaction?

Use Propagation.NEVER - Must execute non-transactionally, throws exception if transaction exists.

Theory:
  • Throws exception if called within transaction
  • Guarantees no transactional context
  • Strict enforcement of non-transactional execution
Use Case:

Reports generation, statistical calculations, read-only operations that should never be in transaction.

@Service
public class ReportService {
    
    @Transactional(propagation = Propagation.NEVER)
    public Report generateComplexReport(Date from, Date to) {
        // Throws IllegalTransactionStateException if called within transaction
        // Guaranteed to run without transaction
        
        // Complex read-only operations
        List<Order> orders = orderRepository.findBetweenDates(from, to);
        List<Customer> customers = customerRepository.findAllActive();
        
        return ReportGenerator.generate(orders, customers);
    }
}

    // This will THROW EXCEPTION:

4. How to require existing transaction (fail if none)?

Use Propagation.MANDATORY - Must have existing transaction, throws exception otherwise.

Theory:
  • Requires existing transaction context
  • Throws IllegalTransactionStateException if no transaction
  • Designed to be part of larger transactional unit
Use Case:

Payment processing, audit trails, or any operation that should only execute as part of a larger business transaction.

@Service
public class PaymentService {
    
    // This should ONLY be called as part of a larger transaction
    @Transactional(propagation = Propagation.MANDATORY)
    public PaymentRecord processPayment(PaymentRequest request) {

        // Will fail if called without transaction context
        
        PaymentRecord record = paymentGateway.charge(request);
        paymentRepository.save(record);
        auditRepository.logPayment(record);
        
        return record;
    }
}

@Service
public class OrderService {
    
    // CORRECT: Method creates transaction
    @Transactional
    public Order completeOrder(Order order, PaymentRequest payment) {

        orderRepository.save(order);
        
        // OK: Called within existing transaction
        paymentService.processPayment(payment);

    public Order createOrderWithoutTransaction(Order order, PaymentRequest payment) {

5. How to execute within nested transaction?

Use Propagation.NESTED - Creates savepoint within existing transaction.

Theory:
  • Creates savepoint in existing transaction
  • Nested failure rolls back to savepoint only
  • Parent failure rolls back everything
  • Requires savepoint support in database
Use Case:

Batch processing, item-by-item processing where individual failures shouldn't abort entire batch.

@Service
public class BatchImportService {
    
    @Transactional
    public ImportResult importProducts(List<ProductDTO> products) {
        ImportResult result = new ImportResult();
        
        for (ProductDTO productDTO : products) {
            try {
                // Each product imported in nested transaction
                importSingleProduct(productDTO);
                result.addSuccess(productDTO.getId());
            } catch (ImportException e) {
                // Only this product's changes rolled back
                // Batch continues with next product
                result.addFailure(productDTO.getId(), e.getMessage());
                log.warn("Failed to import product {}", productDTO.getId(), e);
            }
        }
        
        // All successful imports committed together
        return result;
    }
    
    @Transactional(propagation = Propagation.NESTED)
    public void importSingleProduct(ProductDTO productDTO) throws ImportException {
        // Creates savepoint in parent transaction
        
        // Validate product
        if (!productValidator.isValid(productDTO)) {
            throw new ImportException("Invalid product data");
        }
        
        // Save product
        Product product = convertToEntity(productDTO);
        productRepository.save(product);
        
        // Save product details
        productDetailRepository.save(product.getDetails());
        
        // If any exception thrown here:
        // Rolls back to savepoint (only this product's changes)
        // Parent transaction continues
    }
}

  • Ideal for read operations

Transaction Isolation Control in Spring

1. How to prevent dirty reads?

Theory: Dirty reads occur when a transaction reads uncommitted data from another transaction. To prevent this, use at least READ_COMMITTED isolation level.

Solution with @Transactional

@Service
public class AccountService {
    
    // READ_COMMITTED prevents dirty reads
    @Transactional(isolation = Isolation.READ_COMMITTED)
    public BigDecimal getAccountBalance(Long accountId) {
        // This will only see committed data
        // Won't see uncommitted transfers from other transactions
        return accountRepository.findBalance(accountId);
    }
    
    @Transactional(isolation = Isolation.READ_COMMITTED)
    public void transferMoney(Long fromId, Long toId, BigDecimal amount) {
        // Other transactions won't see these changes until commit
        accountRepository.debit(fromId, amount);
        accountRepository.credit(toId, amount);
    }
}

Database-specific behavior

MySQL

READ_COMMITTED uses consistent reads with MVCC

PostgreSQL

READ_COMMITTED is default, uses snapshot isolation

Oracle

READ_COMMITTED is default and only supported level

Note: READ_COMMITTED prevents dirty reads but allows non-repeatable reads and phantom reads.

2. How to prevent non-repeatable reads?

Theory: Non-repeatable reads occur when a transaction reads the same row twice and gets different values. Use REPEATABLE_READ isolation level.

@Service
public class InventoryService {
    
    // REPEATABLE_READ prevents non-repeatable reads
    @Transactional(isolation = Isolation.REPEATABLE_READ)
    public void validateAndProcessOrder(Order order) {
        // First read
        Integer stock1 = inventoryRepository.getStock(order.getItemId());
        
        // Another transaction CANNOT update this row here
        // Any update would be blocked until this transaction completes
        
        // Second read - guaranteed to get same value
        Integer stock2 = inventoryRepository.getStock(order.getItemId());
        // stock1 == stock2 (guaranteed)
        
        if (stock1 >= order.getQuantity()) {
            inventoryRepository.reduceStock(order.getItemId(), order.getQuantity());
        }
    }
}

// Problem scenario prevented:
// Transaction A: SELECT stock FROM inventory WHERE id = 1 → returns 100
// Transaction B: UPDATE inventory SET stock = 80 WHERE id = 1 → COMMIT
// Transaction A: SELECT stock FROM inventory WHERE id = 1 → STILL returns 100 (not 80)

Database Implementation

  • MySQL InnoDB: Uses MVCC with consistent snapshots
  • SQL Server: Uses shared locks on read data
  • PostgreSQL: Uses snapshot isolation (REPEATABLE_READ)

Limitation

REPEATABLE_READ still allows phantom reads (new rows can appear).

3. How to prevent phantom reads?

Theory: Phantom reads occur when a transaction sees new rows inserted by another transaction. Use SERIALIZABLE isolation level (or MySQL's REPEATABLE_READ with gap locks).

Using SERIALIZABLE

@Service
public class SeatReservationService {
    
    // SERIALIZABLE prevents phantom reads
    @Transactional(isolation = Isolation.SERIALIZABLE)
    public List<Seat> findAndReserveSeats(Show show, int count) {
        // This query locks the range to prevent new seats
        List<Seat> availableSeats = seatRepository
            .findAvailableSeats(show.getId(), count);
        
        // No other transaction can insert seats in this range
        // until this transaction completes
        
        for (Seat seat : availableSeats) {
            seat.reserve();
            seatRepository.save(seat);
        }
        
        return availableSeats;
    }
}

// Phantom read scenario prevented:
// Transaction A: SELECT * FROM seats WHERE show_id = 1 AND status = 'AVAILABLE' → returns 5 seats
// Transaction B: INSERT INTO seats (show_id, status) VALUES (1, 'AVAILABLE') → BLOCKED
// Transaction A: SELECT * FROM seats WHERE show_id = 1 AND status = 'AVAILABLE' → still 5 seats

Alternative: Range Locks (MySQL specific)

@Service
public class ProductService {
    
    // MySQL's REPEATABLE_READ with gap locks prevents phantoms
    @Transactional(isolation = Isolation.REPEATABLE_READ)
    public void applyDiscountToCategory(String category, BigDecimal discount) {
        // MySQL places gap locks to prevent insertions in range
        List<Product> products = productRepository
            .findByCategoryAndPriceBetween(
                category, 
                new BigDecimal("10.00"), 
                new BigDecimal("100.00")
            );

        // for this category until transaction completes

            product.applyDiscount(discount);
            productRepository.save(product);
        }
    }
}

Advanced Transaction Rollback Scenarios

1. How to Rollback Transaction on Checked Exceptions?

By default, Spring only rolls back on RuntimeException and Error. To rollback on checked exceptions, use rollbackFor attribute.

Single Checked Exception

@Service
public class FileImportService {
    
    @Transactional(rollbackFor = IOException.class)
    public void importDataFromFile(String filePath) throws IOException {
        // 1. Read file (might throw IOException)
        String data = fileReader.readFile(filePath);
        
        // 2. Process and save to database
        databaseService.saveData(processData(data));
        
        // 3. If IOException occurs here, transaction rolls back
        fileReader.cleanup(filePath);
    }
}

Multiple Checked Exceptions

@Service
public class PaymentService {
    
    @Transactional(rollbackFor = {
        SQLException.class,
        PaymentGatewayException.class,
        ValidationException.class
    })
    public Payment processPayment(PaymentRequest request) 
            throws SQLException, PaymentGatewayException {
        
        // 1. Validate (checked exception)
        validatePayment(request);
        
        // 2. Save to DB (SQLException possible)
        Payment payment = paymentRepository.save(request);
        
        // 3. Call external service (checked exception)
        gatewayResponse = paymentGateway.process(request);

        payment.updateStatus(gatewayResponse);
        return paymentRepository.save(payment);
    }
}

2. How to Avoid Rollback on Runtime Exceptions?

Use noRollbackFor attribute to prevent rollback on specific runtime exceptions.

Business Exception No Rollback

@Service
public class UserRegistrationService {
    
    @Transactional(noRollbackFor = {
        DuplicateUserException.class,
        WeakPasswordException.class
    })
    public User registerUser(UserDTO userDto) {
        // 1. Check if user exists (business rule)
        if (userRepository.existsByEmail(userDto.getEmail())) {
            // This RuntimeException won't trigger rollback
            throw new DuplicateUserException("Email already exists");
        }
        
        // 2. Validate password strength
        if (!passwordValidator.isStrong(userDto.getPassword())) {
            // No rollback - we want to log failed attempts
            throw new WeakPasswordException("Password too weak");
        }
        
        // 3. Only SQL/data exceptions cause rollback
        User user = convertToEntity(userDto);
        return userRepository.save(user);
    }
}

Notification Exception Handling

@Service
public class OrderService {
    
    @Transactional(noRollbackFor = NotificationException.class)
    public Order createOrder(OrderRequest request) {
        // 1. Save order (must persist even if notification fails)

            // Transaction continues normally
        }

        return order;
    }
}

Programmatic rollback control without throwing exceptions.

Method 1: Using TransactionStatus.setRollbackOnly()

@Service

Global exception handling with transaction awareness.

@ControllerAdvice
@Transactional  // Can be transactional itself
public class GlobalExceptionHandler {
    
    private final RollbackLogger rollbackLogger;
    private final CompensationService compensationService;
    
    // Handle exceptions AFTER transaction rollback
    @ExceptionHandler(DataAccessException.class)
    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public ResponseEntity<ErrorResponse> handleDataAccessException(
            DataAccessException ex, 
            WebRequest request) {
        
        // Transaction has already rolled back by this point
        rollbackLogger.logRollback(ex, request);
        
        return ResponseEntity
            .status(HttpStatus.INTERNAL_SERVER_ERROR)
            .body(new ErrorResponse("Database error occurred", 
                                    request.getDescription(false)));
    }
    
    // Handle business exceptions (no rollback cases)
    @ExceptionHandler(BusinessValidationException.class)
    @ResponseStatus(HttpStatus.BAD_REQUEST)
    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public ResponseEntity<ErrorResponse> handleBusinessException(
            BusinessValidationException ex,
            HttpServletRequest request) {
        
        // This runs in NEW transaction - safe to write to DB
        auditService.logValidationFailure(ex, request);
        
        return ResponseEntity
            .status(HttpStatus.BAD_REQUEST)
            .body(new ErrorResponse(ex.getMessage(), 
                                    request.getRequestURI()));
    }
    
    // Compensation logic after rollback
    @ExceptionHandler(OrderProcessingException.class)
    @ResponseStatus(HttpStatus.CONFLICT)
    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public ResponseEntity<ErrorResponse> handleOrderException(
            OrderProcessingException ex,
            @RequestParam(required = false) Long orderId) {
        
        // Run compensation in new transaction
        if (orderId != null) {
            compensationService.compensateOrder(orderId, ex);
        }
        
        return ResponseEntity
            .status(HttpStatus.CONFLICT)
            .body(new ErrorResponse("Order processing failed", 
                                    "Please retry or contact support"));
    }
    
    // Detect if transaction was rolled back
    @ExceptionHandler(Exception.class)
    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public ResponseEntity<ErrorResponse> handleGenericException(
            Exception ex, 
            HttpServletRequest request) {
        
        // Check transaction status
        try {
            TransactionStatus status = TransactionAspectSupport
                .currentTransactionStatus();
            
            if (status.isRollbackOnly()) {
                log.warn("Transaction marked for rollback", ex);
            }
        } catch (Exception txEx) {
            // No transaction context
        }
        
        return ResponseEntity
            .status(HttpStatus.INTERNAL_SERVER_ERROR)
            .body(new ErrorResponse("Internal server error", 
                                    request.getRequestURI()));
    }
}

@Service

        log.error("All retries failed for payment {}", request.getId(), e);
        
        // Save failure record
        PaymentFailure failure = new PaymentFailure(request, e);
        failureRepository.save(failure);
        
        // Compensate
        compensationService.compensatePayment(request);

    }

Method 2: Manual Retry with Circuit Breaker

@Service
public class OrderRetryService {
    
    private final CircuitBreaker circuitBreaker;
    
    public OrderRetryService() {
        this.circuitBreaker = CircuitBreaker.ofDefaults("orderService");
    }
    
    @Transactional
    public Order processWithManualRetry(OrderRequest request) {
        
        int maxRetries = 3;
        int attempt = 0;
        
        while (attempt < maxRetries) {
            attempt++;
            
            try {
                // Wrap with circuit breaker
                return circuitBreaker.executeSupplier(() -> {
                    
                    // Start new transaction for each retry
                    return TransactionTemplate.execute(status -> {
                        
                        // Save order with retry tracking
                        Order order = createOrder(request);
                        order.setRetryAttempt(attempt);
                        order = orderRepository.save(order);
                        
                        // Try inventory reservation
                        inventoryService.reserveWithLock(order);
                        
                        // Try payment
                        paymentService.process(order.getPayment());
                        
                        order.confirm();
                        return orderRepository.save(order);
                    });
                });
                
            } catch (OptimisticLockingFailureException e) {
                // Concurrent modification - retry
                log.warn("Optimistic lock failure, attempt {}", attempt);
                
                if (attempt == maxRetries) {
                    throw new OrderFailedException("Max retries exceeded", e);
                }
                
                // Exponential backoff
                sleep(exponentialBackoff(attempt));
                continue;
                
            } catch (Exception e) {
                // Check if retryable
                if (isRetryable(e) && attempt < maxRetries) {
                    log.warn("Retryable error, attempt {}", attempt, e);
                    
                    // Rollback current transaction
                    TransactionAspectSupport.currentTransactionStatus()
                        .setRollbackOnly();
                    
                    sleep(1000 * attempt); // Linear backoff
                    continue;
                }
                
                // Non-retryable or max attempts reached
                throw new OrderFailedException("Order processing failed", e);
            }
        }
        
        throw new OrderFailedException("Max retry attempts reached");
    }
    
    private boolean isRetryable(Exception e) {
        return e instanceof TransientDataAccessException ||
               e instanceof DeadlockLoserDataAccessException ||
               e instanceof SocketTimeoutException ||
               (e.getMessage() != null && 
                e.getMessage().contains("timeout"));
    }
    
    private void sleep(long millis) {
        try {
            Thread.sleep(millis);
        } catch (InterruptedException ie) {
            Thread.currentThread().interrupt();
        }
    }
}

Method 3: Retry Template Pattern

@Component
public class RetryTemplateService {
    
    private final RetryTemplate retryTemplate;
    
    public RetryTemplateService() {
        this.retryTemplate = RetryTemplate.builder()
            .maxAttempts(3)
            .exponentialBackoff(1000, 2, 10000)
            .retryOn(TransientDataAccessException.class)
            .retryOn(SocketTimeoutException.class)
            .notRetryOn(PermanentFailureException.class)
            .withListener(new RetryListener() {
                @Override
                public <T, E extends Throwable> void onError(
                    RetryContext context, 
                    RetryCallback<T, E> callback, 
                    Throwable throwable) {
                    
                    log.warn("Retry attempt {} for {}", 
                        context.getRetryCount(), 
                        throwable.getMessage());
                }
            })
            .build();
    }
    
    @Transactional
    public Result processWithRetryTemplate(Input input) {
        
        return retryTemplate.execute(context -> {
            
            // Each retry runs in new transaction
            return TransactionTemplate.execute(status -> {
                
                // Track retry attempt
                int attempt = context.getRetryCount() + 1;
                log.info("Processing attempt {}", attempt);
                
                // Business logic that might fail
                Result result = businessService.process(input);
                
                // If we reach here, success - commit
                return result;
                
            }, transactionTemplate);
            
        }, recoveryCallback);
    }
    
    private final RecoveryCallback<Result> recoveryCallback = context -> {
        // Called after all retries fail
        log.error("All retries failed after {} attempts", 
            context.getRetryCount());
        
        // Execute compensation in new transaction
        return TransactionTemplate.execute(status -> {
            compensationService.compensate(context.getLastThrowable());
            return Result.failure("All retries exhausted");
        });
    };
}

            accountRepository.debit(fromId, amount);
            
            // Additional validation
            if (amount.compareTo(MAX_LIMIT) > 0) {
                // Manual rollback decision
                status.setRollbackOnly();
                throw new LimitExceededException("Amount exceeds limit");
            }
            
            accountRepository.credit(toId, amount);
            auditRepository.logTransfer(fromId, toId, amount);
            
            // 4. Commit transaction
            transactionManager.commit(status);
            
        } catch (Exception e) {
            // 5. Rollback on any exception
            if (!status.isCompleted()) {
                transactionManager.rollback(status);
            }
            throw new BusinessException("Transfer failed", e);
        }
    }
    
    // Multiple transactions in same method
    public void complexOperation(Data data) {
        // Transaction 1: Save main entity
        DefaultTransactionDefinition tx1Def = new DefaultTransactionDefinition();
        tx1Def.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRED);
        
        TransactionStatus tx1Status = transactionManager.getTransaction(tx1Def);
        try {
            mainRepository.save(data.getMainEntity());
            transactionManager.commit(tx1Status);
        } catch (Exception e) {
            transactionManager.rollback(tx1Status);
            throw e;
        }
        
        // Transaction 2: Save details in separate transaction
        DefaultTransactionDefinition tx2Def = new DefaultTransactionDefinition();
        tx2Def.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRES_NEW);
        
        TransactionStatus tx2Status = transactionManager.getTransaction(tx2Def);
        try {
            for (Detail detail : data.getDetails()) {
                detailRepository.save(detail);
            }
            transactionManager.commit(tx2Status);
        } catch (Exception e) {
            transactionManager.rollback(tx2Status);
            log.error("Failed to save details", e);
            // Don't rethrow - first transaction already committed
        }
    }
    
    // Check transaction status
    public boolean isTransactionActive() {
        return TransactionSynchronizationManager.isActualTransactionActive();
    }
    
    public boolean isTransactionReadOnly() {
        return TransactionSynchronizationManager.isCurrentTransactionReadOnly();
    }
}

@Component
public class TransactionContextTaskDecorator implements TaskDecorator {
    
    @Override
    public Runnable decorate(Runnable runnable) {
        // Capture current transaction context
        Map resources = TransactionSynchronizationManager.getResourceMap();
        boolean actualTransactionActive = TransactionSynchronizationManager.isActualTransactionActive();
        String transactionName = TransactionSynchronizationManager.getCurrentTransactionName();
        boolean readOnly = TransactionSynchronizationManager.isCurrentTransactionReadOnly();
        Integer isolationLevel = TransactionSynchronizationManager.getCurrentTransactionIsolationLevel();
        
        return () -> {
            try {
                // Restore transaction context in async thread
                TransactionSynchronizationManager.setActualTransactionActive(actualTransactionActive);
                TransactionSynchronizationManager.setCurrentTransactionName(transactionName);
                TransactionSynchronizationManager.setCurrentTransactionReadOnly(readOnly);
                if (isolationLevel != null) {
                    TransactionSynchronizationManager.setCurrentTransactionIsolationLevel(isolationLevel);
                }
                
                // Set resources
                resources.forEach((key, value) -> 
                    TransactionSynchronizationManager.bindResource(key, value)
                );
                
                runnable.run();
            } finally {
                // Clean up
                resources.keySet().forEach(
                    TransactionSynchronizationManager::unbindResource
                );
                TransactionSynchronizationManager.clear();
            }
        };
    }

7. How to Use @Transactional with CompletableFuture?

@Service
public class CompletableFutureTransactionService {
    
    @Autowired
    private PlatformTransactionManager transactionManager;
    
    @Autowired
    private OrderRepository orderRepository;
    
    @Autowired
    private PaymentRepository paymentRepository;
    
    // Pattern 1: Transactional method returning CompletableFuture
    @Transactional
    public CompletableFuture createOrderAsync(OrderRequest request) {
        // This part runs in transaction
        Order order = orderRepository.save(convertToOrder(request));
        
        // Return CompletableFuture for async continuation
        return CompletableFuture.completedFuture(order)
            .thenApplyAsync(savedOrder -> {
                // This runs outside original transaction!
                // Start new transaction if needed
                TransactionTemplate txTemplate = new TransactionTemplate(transactionManager);
                return txTemplate.execute(status -> {
                    // Async processing in new transaction
                    processPostCreation(savedOrder);
                    return savedOrder;
                });
            });
    }
    
    // Pattern 2: Chain multiple transactional operations
    public CompletableFuture processOrderPipeline(OrderRequest request) {
        return CompletableFuture.supplyAsync(() -> {
            // Step 1: Create order (transaction 1)
            return executeInTransaction(() -> 
                orderRepository.save(convertToOrder(request))
            );
        })
        .thenApplyAsync(order -> {
            // Step 2: Process payment (transaction 2)  
            return executeInTransaction(() -> {
                Payment payment = processPayment(order);
                order.setPayment(payment);
                return orderRepository.save(order);
            });
        })
        .thenApplyAsync(order -> {
            // Step 3: Update inventory (transaction 3)
            return executeInTransaction(() -> {
                updateInventory(order);
                order.setStatus(OrderStatus.COMPLETED);
                return orderRepository.save(order);
            });
        })
        .exceptionally(throwable -> {
            // Handle failure in any step
            log.error("Order processing failed", throwable);
            throw new BusinessException("Processing failed", throwable);
        });
    }
    
    // Pattern 3: Parallel transactional operations
    public CompletableFuture processInParallel(Order order) {
        CompletableFuture orderFuture = CompletableFuture.supplyAsync(() ->
            executeInTransaction(() -> orderRepository.save(order))
        );
        
        CompletableFuture paymentFuture = CompletableFuture.supplyAsync(() ->
            executeInTransaction(() -> paymentRepository.process(order))
        );
        
        CompletableFuture inventoryFuture = CompletableFuture.supplyAsync(() ->
            executeInTransaction(() -> inventoryRepository.reserve(order))
        );
        
        // Combine all results
        return CompletableFuture.allOf(orderFuture, paymentFuture, inventoryFuture)
            .thenApplyAsync(voidResult -> {
                // Combine results in new transaction
                return executeInTransaction(() -> {
                    Order savedOrder = orderFuture.join();
                    Payment payment = paymentFuture.join();
                    Inventory inventory = inventoryFuture.join();
                    
                    return new OrderResult(savedOrder, payment, inventory);
                });
            });
    }
    
    // Helper method to execute in transaction
    private  T executeInTransaction(Supplier supplier) {
        TransactionTemplate txTemplate = new TransactionTemplate(transactionManager);
        txTemplate.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRED);
        
        return txTemplate.execute(status -> supplier.get());
    }
    
    // Pattern 4: Transactional with timeout handling
    @Transactional(timeout = 30)
    public CompletableFuture createOrderWithTimeout(OrderRequest request) {
        return CompletableFuture.supplyAsync(() -> {
            Order order = orderRepository.save(convertToOrder(request));
            
            // Chain async operations that respect transaction timeout
            return order;
        })
        .orTimeout(25, TimeUnit.SECONDS)  // Less than transaction timeout
        .exceptionally(throwable -> {
            if (throwable instanceof TimeoutException) {
                // Transaction will rollback due to timeout
                throw new TransactionTimedOutException("Operation timed out");
            }
            throw new CompletionException(throwable);
        });
    }
    
    // Pattern 5: Transaction event listeners with CompletableFuture
    @Transactional
    public CompletableFuture createOrderWithEvents(OrderRequest request) {
        Order order = orderRepository.save(convertToOrder(request));
        
        // Create event that will be processed after commit
        applicationEventPublisher.publishEvent(new OrderCreatedEvent(this, order.getId()));
        
        return CompletableFuture.completedFuture(order);
    }
    
    @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
    public void handleAfterCommit(OrderCreatedEvent event) {
        // This runs after transaction commits
        CompletableFuture.runAsync(() -> {
            // Async processing after successful commit
            notificationService.sendConfirmation(event.getOrderId());
            analyticsService.trackOrder(event.getOrderId());
        });
    }
}

⚠️ Key Considerations

  • Thread Boundaries: Transactions are thread-bound, async methods lose transaction context
  • Propagation: Understand REQUIRED vs REQUIRES_NEW for method chaining
  • Self-Invocation: Use self-injection or AspectJ mode for internal calls
  • Timeout: Transaction timeouts don't apply to async operations after method returns
  • Error Handling: Handle exceptions properly in async transactional flows
  • Resource Cleanup: Always clean up TransactionSynchronizationManager in async threads

        Order savedOrder = orderRepository.save(order);
        
        // 2. Log action in secondary DB
        auditService.logAction("ORDER_CREATED", "Order:" + savedOrder.getId());

        // BUT they are NOT atomic together!

    public void processOrderManual(Order order) {

        TransactionStatus primaryStatus = primaryTransactionManager.getTransaction(def);
        
        try {
            // Save order
            Order savedOrder = orderRepository.save(order);
            
            // Start secondary transaction
            TransactionStatus secondaryStatus = secondaryTransactionManager.getTransaction(def);
            
            try {
                // Log audit
                auditService.logAction("ORDER_CREATED", "Order:" + savedOrder.getId());
                secondaryTransactionManager.commit(secondaryStatus);
                
            } catch (Exception e) {
                secondaryTransactionManager.rollback(secondaryStatus);
                primaryTransactionManager.rollback(primaryStatus);
                throw e;
            }
            
            primaryTransactionManager.commit(primaryStatus);
            
        } catch (Exception e) {
            primaryTransactionManager.rollback(primaryStatus);
            throw e;
        }
    }
}

        payment.setOrderId(savedOrder.getId());
        Payment processedPayment = paymentRepository.save(payment);
        
        // 3. Update inventory in Database 3
        inventoryRepository.reduceStock(order.getItems());
        
        // All three operations succeed or fail together
        // JTA coordinates the two-phase commit across all databases
    }
}

Saga Pattern (Alternative to JTA)

@Service
public class OrderSagaService {
    
    private final SagaProcessor sagaProcessor;
    
    @Transactional
    public Order processOrderSaga(OrderRequest request) {
        return sagaProcessor.execute(() -> {
            // Step 1: Create order (local transaction)
            Order order = orderService.createOrder(request.getOrder());

            inventoryService.reserveItems(order.getItems());
            
            // Step 3: Process payment (remote service - separate transaction)  
            paymentService.processPayment(order.getPayment());
            
            // Step 4: Confirm order
            order.confirm();
            orderRepository.save(order);
            
            return order;
            
        }, (order, exception) -> {
            // Compensation logic if any step fails
            log.error("Saga failed for order {}", order.getId(), exception);
            
            try {
                // Reverse inventory reservation
                inventoryService.releaseItems(order.getItems());
            } catch (Exception e) {
                log.error("Failed to release inventory", e);
            }
            
            try {
                // Refund payment if processed
                if (order.getPayment().isProcessed()) {
                    paymentService.refund(order.getPayment());
                }
            } catch (Exception e) {
                log.error("Failed to refund payment", e);
            }
            
            // Mark order as failed
            order.fail(exception.getMessage());
            orderRepository.save(order);
        });
    }
}

            return action.get();

5. How to handle savepoints in transactions?

Savepoint with Programmatic Transactions

@Service
public class SavepointService {
    
    @Autowired
    private PlatformTransactionManager transactionManager;
    
    public void processWithSavepoints(List<Item> items) {
        DefaultTransactionDefinition def = new DefaultTransactionDefinition();
        TransactionStatus status = transactionManager.getTransaction(def);
        
        try {
            for (int i = 0; i < items.size(); i++) {
                // Create savepoint before each item
                Object savepoint = status.createSavepoint();
                
                try {
                    processItem(items.get(i));
                    
                } catch (ItemProcessingException e) {
                    // Rollback to savepoint - undo only this item
                    status.rollbackToSavepoint(savepoint);
                    log.error("Failed to process item {}, continuing with next", i, e);
                    continue;
                    
                } finally {
                    // Release savepoint
                    status.releaseSavepoint(savepoint);
                }
            }
            
            transactionManager.commit(status);
            
        } catch (Exception e) {
            transactionManager.rollback(status);
            throw e;
        }
    }
    
    private void processItem(Item item) {
        // Item processing logic
        itemRepository.save(item);
        inventoryService.update(item);
        auditService.log(item);
    }
}

6. How to flush and clear in JPA transactions?

Understanding Flush and Clear

Flush Behavior:
  • Automatic flush: Before query execution and transaction commit
  • Manual flush: entityManager.flush() forces sync with database
  • FlushMode: AUTO (default) or COMMIT (only at commit)
Clear Behavior:
  • Detaches all entities from persistence context
  • Forces reload from database on next access
  • Useful for batch processing to prevent memory issues

Batch Processing with Flush and Clear

@Service
public class BatchImportService {
    
    @PersistenceContext
    private EntityManager entityManager;
    
    @Transactional
    public void importLargeDataset(List<Product> products) {
        int batchSize = 50;
        
        for (int i = 0; i < products.size(); i++) {
            Product product = products.get(i);
            
            // 1. Persist entity
            entityManager.persist(product);
            
            // 2. Flush and clear at batch interval
            if (i > 0 && i % batchSize == 0) {
                // Force write to database
                entityManager.flush();
                
                // Clear persistence context to free memory
                entityManager.clear();
                
                log.info("Processed {} of {} products", i, products.size());
            }
        }
        
        // Final flush for remaining entities
        entityManager.flush();
    }
    
    @Transactional
    public void batchUpdatePrices(List<Long> productIds, BigDecimal increasePercent) {
        int batchSize = 100;
        
        for (int i = 0; i < productIds.size(); i++) {
            Long productId = productIds.get(i);
            
            // 1. Find entity (loads into persistence context)
            Product product = entityManager.find(Product.class, productId);
            
            // 2. Update entity
            if (product != null) {
                BigDecimal newPrice = product.getPrice()
                    .multiply(BigDecimal.ONE.add(increasePercent));
                product.setPrice(newPrice);
            }
            
            // 3. Flush and clear periodically
            if (i > 0 && i % batchSize == 0) {
                // Write changes to database
                entityManager.flush();
                
                // Detach processed entities
                entityManager.clear();
            }
        }
    }
}

3. How to monitor transaction performance?

Theory:

Monitoring helps identify bottlenecks, deadlocks, and performance issues:

  • Transaction duration: Time from start to commit
  • Rollback rate: Percentage of failed transactions
  • Concurrent transactions: Active transaction count
  • Lock wait time: Time spent waiting for locks
  • Connection usage: Active vs idle connections

// 1. Aspect for transaction monitoring
@Aspect
@Component
@Slf4j
public class TransactionMonitoringAspect {
    
    private final MeterRegistry meterRegistry;
    private final Map<String, Long> startTimes = new ConcurrentHashMap<>();
    
    public TransactionMonitoringAspect(MeterRegistry meterRegistry) {
        this.meterRegistry = meterRegistry;
    }
    
    @Around("@annotation(transactional)")
    public Object monitorTransaction(ProceedingJoinPoint joinPoint, 
                                     Transactional transactional) throws Throwable {
        
        String methodName = joinPoint.getSignature().toShortString();
        String transactionKey = UUID.randomUUID().toString();
        
        startTimes.put(transactionKey, System.currentTimeMillis());
        meterRegistry.counter("transaction.started", "method", methodName).increment();
        
        try {
            Object result = joinPoint.proceed();
            
            long duration = System.currentTimeMillis() - startTimes.get(transactionKey);
            meterRegistry.timer("transaction.success", "method", methodName)
                        .record(duration, TimeUnit.MILLISECONDS);
            meterRegistry.gauge("transaction.duration", duration);
            
            log.info("Transaction completed for {} in {} ms", methodName, duration);
            return result;
            
        } catch (Exception e) {
            meterRegistry.counter("transaction.failed", "method", methodName, 
                                 "exception", e.getClass().getSimpleName()).increment();
            log.error("Transaction failed for {}: {}", methodName, e.getMessage());
            throw e;
        } finally {
            startTimes.remove(transactionKey);
        }
    }
}

4. How to handle deadlocks in transactions?

Theory:

Deadlocks occur when transactions wait for each other's locks:

  • Detection: Database detects and rolls back one transaction
  • Prevention: Consistent access order, short transactions
  • Retry: Automatically retry on deadlock exceptions
  • Timeout: Use lock timeouts instead of waiting indefinitely

Code Implementation:

// 1. Deadlock retry aspect
@Aspect
@Component
@Slf4j
public class DeadlockRetryAspect {
    
    private static final int MAX_RETRIES = 3;
    private static final long RETRY_DELAY_MS = 100;
    
    @Around("@annotation(transactional)")
    public Object retryOnDeadlock(ProceedingJoinPoint joinPoint, 
                                  Transactional transactional) throws Throwable {
        
        int attempt = 0;
        while (attempt < MAX_RETRIES) {
            try {
                return joinPoint.proceed();
                
            } catch (Exception e) {
                if (isDeadlockException(e)) {
                    attempt++;
                    log.warn("Deadlock detected on attempt {} for method {}", 
                            attempt, joinPoint.getSignature().getName());
                    
                    if (attempt < MAX_RETRIES) {
                        // Exponential backoff
                        long delay = RETRY_DELAY_MS * (long) Math.pow(2, attempt - 1);
                        Thread.sleep(delay);
                        continue;
                    }
                }
                throw e;
            }
        }
        throw new RuntimeException("Max retries exceeded");
    }
    
    private boolean isDeadlockException(Exception e) {
        return e.getCause() != null && (
            e.getCause().getMessage().contains("deadlock") ||
            e.getCause().getMessage().contains("lock wait timeout") ||
            e.getCause() instanceof CannotAcquireLockException ||
            e.getCause() instanceof PessimisticLockingFailureException
        );
    }
}

5. How to batch operations within a transaction?

Theory:

Batching reduces database round-trips and improves performance:

  • JPA Batch: Group multiple operations into single SQL statements
  • JDBC Batch: Use PreparedStatement.addBatch()
  • Chunk Processing: Process large datasets in chunks
  • Memory Management: Clear persistence context periodically

// 1. JPA Batch Insert/Update

    private <T> List<List<T>> partition(List<T> list, int size) {
        List<List<T>> partitions = new ArrayList<>();
        for (int i = 0; i < list.size(); i += size) {
            partitions.add(list.subList(i, Math.min(i + size, list.size())));
        }
        return partitions;
    }

6. How to reduce transaction scope for performance?

Theory:

Smaller transactions improve concurrency and reduce lock contention:

  • Keep transactions short: Do work quickly, commit early
  • External calls outside: Move API calls, file I/O outside transactions
  • Split large transactions: Break into smaller units
  • Use read-only when possible: Avoid unnecessary write locks
  • Commit frequently: In batch processing, commit per chunk

Code Implementation:

        def.setName("ManualTransferTx");

        def.setIsolationLevel(TransactionDefinition.ISOLATION_READ_COMMITTED);

        TransactionStatus status = transactionManager.getTransaction(def);

            accountRepository.debit(fromId, amount);

        metricsService.incrementOrderProcessed();

5. How to Implement Compensating Transactions (Saga Pattern)

Long-running distributed transactions with compensating actions for rollback.

// Saga Coordinator
@Service
public class OrderSagaService {
    
    @Autowired
    private OrderRepository orderRepository;
    
    @Autowired
    private InventoryService inventoryService;
    
    @Autowired
    private PaymentService paymentService;
    
    @Autowired
    private NotificationService notificationService;
    
    @Transactional
    public Order createOrderWithSaga(OrderRequest request) {
        Order order = null;
        
        try {
            // Step 1: Create order (local transaction)
            order = createOrderLocal(request);
            
            // Step 2: Reserve inventory (remote - might fail)
            inventoryService.reserveItems(order.getItems());
            
            // Step 3: Process payment (remote - might fail)
            paymentService.processPayment(order.getPayment());
            
            // Step 4: Finalize order
            order.confirm();
            orderRepository.save(order);
            
            // Step 5: Send notifications (non-critical)
            notificationService.sendOrderConfirmation(order);
            
            return order;
            
        } catch (Exception e) {
            // Compensation: Reverse all completed steps
            if (order != null) {
                compensate(order, e);
            }
            throw new SagaException("Order creation failed", e);
        }
    }
    
    private void compensate(Order order, Exception failureCause) {
        log.error("Compensating saga for order {} due to: {}", 
                 order.getId(), failureCause.getMessage());
        
        List<CompensationAction> completedActions = new ArrayList<>();
        
        try {
            // Try to reverse inventory reservation
            if (order.getInventoryReserved()) {
                inventoryService.releaseItems(order.getItems());
                completedActions.add(CompensationAction.INVENTORY_RELEASED);
            }
        } catch (Exception e) {
            log.error("Failed to release inventory for order {}", order.getId(), e);
            // Continue with other compensations
        }
        
        try {
            // Try to reverse payment
            if (order.getPayment().isProcessed()) {
                paymentService.refundPayment(order.getPayment());
                completedActions.add(CompensationAction.PAYMENT_REFUNDED);
            }
        } catch (Exception e) {
            log.error("Failed to refund payment for order {}", order.getId(), e);
            // Continue
        }
        
        // Mark order as failed
        order.fail("Compensated: " + failureCause.getMessage());
        orderRepository.save(order);
        
        log.info("Compensation completed for order {}. Actions: {}", 
                order.getId(), completedActions);
    }
    
    // Alternative: Choreography-based Saga
    @Transactional
    public void processOrderChoreography(Order order) {
        // Each service emits events, other services react
        
        // 1. Order Service: Create order
        orderRepository.save(order);
        eventPublisher.publishEvent(new OrderCreatedEvent(order));
        
        // Inventory Service listens to OrderCreatedEvent → reserves inventory
        // Payment Service listens to InventoryReservedEvent → processes payment
        // Order Service listens to PaymentProcessedEvent → confirms order
        
        // If any service fails, it publishes compensation events
        // Other services listen and perform compensating actions
    }
}

// Compensation tracking
@Entity
public class SagaLog {
    @Id
    private String sagaId;
    private String orderId;
    private SagaStatus status;
    private List<String> completedSteps;
    private List<String> compensationActions;
    private String failureReason;
    
    public enum SagaStatus {
        STARTED, COMPLETED, COMPENSATING, COMPENSATED, FAILED
    }
}

@Service
public class SagaOrchestrator {
    
    @Transactional
    public void executeSaga(String sagaId, List<SagaStep> steps) {
        SagaLog log = new SagaLog(sagaId);
        log.setStatus(SagaStatus.STARTED);
        
        for (SagaStep step : steps) {
            try {
                step.execute();
                log.getCompletedSteps().add(step.getName());
            } catch (Exception e) {
                log.setStatus(SagaStatus.COMPENSATING);
                compensate(log, steps);
                throw new SagaException("Saga failed at step: " + step.getName(), e);
            }
        }
        
        log.setStatus(SagaStatus.COMPLETED);
        sagaLogRepository.save(log);
    }
    
    private void compensate(SagaLog log, List<SagaStep> steps) {
        // Execute compensations in reverse order
        ListIterator<SagaStep> iterator = 
            steps.listIterator(steps.size());
        
        while (iterator.hasPrevious()) {
            SagaStep step = iterator.previous();
            if (log.getCompletedSteps().contains(step.getName())) {
                try {
                    step.compensate();
                    log.getCompensationActions().add("Compensated: " + step.getName());
                } catch (Exception e) {
                    log.getCompensationActions().add("Failed to compensate: " + step.getName());
                    // Continue with other compensations
                }
            }
        }
        
        log.setStatus(SagaStatus.COMPENSATED);
        sagaLogRepository.save(log);
    }
}

// Create a safe wrapper for operations that might set rollback-only
@Component
public class TransactionGuard {
    
    @Autowired
    private PlatformTransactionManager transactionManager;
    
    public <T> T executeInNewTransaction(Supplier<T> operation) {
        DefaultTransactionDefinition def = new DefaultTransactionDefinition();
        def.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRES_NEW);
        
        TransactionStatus status = transactionManager.getTransaction(def);
        
        try {
            T result = operation.get();
            transactionManager.commit(status);
            return result;
        } catch (Exception e) {
            transactionManager.rollback(status);
            throw e;
        }
    }
    
    public void executeInNewTransaction(Runnable operation) {
        executeInNewTransaction(() -> {
            operation.run();
            return null;
        });
    }
}

    }

    @CachePut(value = "products", key = "#result.id")
    @Transactional
    public Product createProduct(Product product) {
        Product saved = productRepository.save(product);
        
        // Register cache update to happen AFTER commit
        TransactionSynchronizationManager.registerSynchronization(
            new TransactionSynchronization() {
                @Override
                public void afterCommit() {
                    // Force cache refresh after commit
                    cacheManager.getCache("products").evict(saved.getId());
                    cacheManager.getCache("products").put(saved.getId(), saved);
                }
            }
        );
        
        return saved;
    }
}

Chunk-based Transaction Management

@Configuration
@EnableBatchProcessing
public class BatchTransactionConfig {
    
    @Autowired
    private JobBuilderFactory jobBuilderFactory;
    
    @Autowired
    private StepBuilderFactory stepBuilderFactory;
    
    @Bean
    public Job importUserJob() {
        return jobBuilderFactory.get("importUserJob")
            .incrementer(new RunIdIncrementer())
            .start(step1())
            .build();
    }
    
    @Bean
    public Step step1() {
        return stepBuilderFactory.get("step1")
            .<User, User>chunk(10)  // Process 10 items per transaction
            .reader(reader())
            .processor(processor())
            .writer(writer())
            .faultTolerant()
            .skipLimit(10)
            .skip(Exception.class)
            .retryLimit(3)
            .retry(DeadlockLoserDataAccessException.class)
            .transactionManager(batchTransactionManager())  // Custom TM for batch
            .build();
    }
    
    @Bean
    public PlatformTransactionManager batchTransactionManager() {
        DataSourceTransactionManager transactionManager = 
            new DataSourceTransactionManager(dataSource());
        transactionManager.setDefaultTimeout(300);  // Longer timeout for batch
        transactionManager.setNestedTransactionAllowed(true);
        return transactionManager;
    }
    
    @Bean
    @StepScope
    public ItemReader<User> reader() {
        JdbcCursorItemReader<User> reader = new JdbcCursorItemReader<>();
        reader.setDataSource(dataSource());
        reader.setSql("SELECT * FROM users WHERE processed = false");
        reader.setRowMapper(new BeanPropertyRowMapper<>(User.class));
        return reader;
    }
    
    @Bean
    @StepScope
    public ItemProcessor<User, User> processor() {
        return user -> {
            // Process within transaction
            user.setProcessed(true);
            user.setProcessedDate(LocalDateTime.now());
            return user;
        };
    }
    
    @Bean
    @StepScope
    public ItemWriter<User> writer() {
        JdbcBatchItemWriter<User> writer = new JdbcBatchItemWriter<>();
        writer.setDataSource(dataSource());
        writer.setSql("UPDATE users SET processed = ?, processed_date = ? WHERE id = ?");
        writer.setItemPreparedStatementSetter((user, ps) -> {
            ps.setBoolean(1, user.isProcessed());
            ps.setTimestamp(2, Timestamp.valueOf(user.getProcessedDate()));
            ps.setLong(3, user.getId());
        });
        return writer;
    }
}

Compensating Transactions Pattern (Saga)

// 1. Saga Coordinator
@Service
@Transactional
public class OrderSagaService {
    
    public Order createDistributedOrder(OrderRequest request) {
        Order order = new Order();
        order.setStatus(OrderStatus.PENDING);
        
        try {
            // Phase 1: Try all operations
            order = tryCreateOrder(request);
            
            // Phase 2: Commit if all succeeded
            confirmOrder(order);
            
            return order;
            
        } catch (Exception e) {
            // Phase 3: Compensate on failure
            compensate(order, e);
            throw e;
        }
    }
    
    private Order tryCreateOrder(OrderRequest request) throws Exception {
        Order order = orderRepository.save(
            createOrderEntity(request)
        );
        
        // Try inventory service
        inventoryService.reserve(request.getItems());
        
        // Try payment service
        paymentService.authorize(request.getPayment());
        
        // Try shipping service
        shippingService.schedule(request.getShippingAddress());
        
        return order;
    }
    
    private void confirmOrder(Order order) {
        order.setStatus(OrderStatus.CONFIRMED);
        orderRepository.save(order);
        
        // Confirm all services
        inventoryService.confirm(order.getItems());
        paymentService.capture(order.getPaymentId());
        shippingService.confirm(order.getId());
    }
    
    private void compensate(Order order, Exception cause) {
        log.error("Compensating order {}: {}", order.getId(), cause.getMessage());
        
        order.setStatus(OrderStatus.FAILED);
        orderRepository.save(order);
        
        // Compensation actions (idempotent)
        try {
            inventoryService.release(order.getItems());
        } catch (Exception e) {
            log.error("Failed to release inventory", e);
        }
        
        try {
            if (order.getPaymentId() != null) {
                paymentService.refund(order.getPaymentId());
            }
        } catch (Exception e) {
            log.error("Failed to refund payment", e);
        }
        
        try {
            shippingService.cancel(order.getId());
        } catch (Exception e) {
            log.error("Failed to cancel shipping", e);
        }
    }
}

Transactional REST Client with Retry

// Transaction-aware REST client
@Component
@Transactional(propagation = Propagation.MANDATORY)
// Must be called within transaction
public class TransactionalRestClient {
    
    @Autowired
    private RestTemplate restTemplate;
    
    @Autowired
    private ExternalCallRepository callRepository;
    
    @Retryable(value = {ResourceAccessException.class}, 
               maxAttempts = 3,
               backoff = @Backoff(delay = 1000))
    public <T> T postWithTransaction(String url, Object request, 
                                   Class<T> responseType) {
        
        // Record external call attempt
        ExternalCall call = new ExternalCall();
        call.setUrl(url);
        call.setRequest(serialize(request));
        call.setStatus(ExternalCallStatus.PENDING);
        callRepository.save(call);
        
        try {
            // Make REST call
            ResponseEntity<T> response = restTemplate.postForEntity(
                url, request, responseType);
            
            // Update call status
            call.setStatus(ExternalCallStatus.SUCCESS);
            call.setResponse(serialize(response.getBody()));
            callRepository.save(call);
            
            return response.getBody();
            
        } catch (Exception e) {
            call.setStatus(ExternalCallStatus.FAILED);
            call.setError(e.getMessage());
            callRepository.save(call);
            throw e;
        }
    }
    
    // Called after transaction commits
    @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
    public void handleSuccessfulTransaction(ExternalCallEvent event) {
        // Send notifications, update external systems, etc.
        notificationService.sendSuccessNotification(event.getCallId());
    }
    
    // Called after transaction rolls back
    @TransactionalEventListener(phase = TransactionPhase.AFTER_ROLLBACK)
    public void handleFailedTransaction(ExternalCallEvent event) {
        // Cleanup external calls that shouldn't have happened
        externalCallCleanupService.cleanupFailedTransaction(event.getCallId());
    }
}
