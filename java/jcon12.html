Java Callable & Future

Complete Guide to Asynchronous Task Execution

Overview Callable Future CompletableFuture Patterns Best Practices

Callable & Future Overview

Callable and Future are key interfaces in Java's concurrency framework that enable asynchronous task execution with return values and exception handling capabilities.

Key Benefits

  • Return values from asynchronous tasks
  • Proper exception handling
  • Task cancellation support
  • Timeout mechanisms
  • Composable asynchronous operations

Execution Flow

Callable

Task that returns a result

ExecutorService

Submits task for execution

Future

Handle to retrieve result

Basic Callable & Future Example

import java.util.concurrent.*;
import java.util.Random;

public class BasicCallableFutureExample {
    
    public static void main(String[] args) {
        // Create executor service
        ExecutorService executor = Executors.newFixedThreadPool(3);
        
        // Submit Callable task
        Future future = executor.submit(new Callable() {
            @Override
            public Integer call() throws Exception {
                // Simulate some computation
                Thread.sleep(2000);
                Random random = new Random();
                int result = random.nextInt(100);
                System.out.println("Task completed with result: " + result);
                return result;
            }
        });
        
        System.out.println("Task submitted, doing other work...");
        
        try {
            // Do other work while task is executing
            Thread.sleep(1000);
            System.out.println("Other work completed");
            
            // Get the result (blocks if not ready)
            Integer result = future.get();
            System.out.println("Final result: " + result);
            
        } catch (InterruptedException | ExecutionException e) {
            e.printStackTrace();
        } finally {
            executor.shutdown();
        }
    }
}

Callable Interface Deep Dive

Callable Interface

Signature:

public interface Callable<V> { V call() throws Exception; }

Key Features:

  • Generic interface - returns result of type V
  • Can throw checked exceptions
  • Used with ExecutorService.submit()
  • Alternative to Runnable with return value
// Different ways to implement Callable

// 1. Traditional class implementation
class TraditionalCallable implements Callable {
    @Override
    public String call() throws Exception {
        return "Result from traditional callable";
    }
}

// 2. Anonymous class
Callable anonymousCallable = new Callable() {
    @Override
    public Integer call() throws Exception {
        return 42;
    }
};

// 3. Lambda expression (Java 8+)
Callable lambdaCallable = () -> {
    Thread.sleep(1000);
    return Math.random();
};

// 4. Method reference
Callable methodRefCallable = this::generateResult;

private String generateResult() {
    return "Result from method";
}

Callable vs Runnable

Aspect Callable Runnable
Return Value Yes (generic type) No (void)
Exception Can throw checked exceptions Cannot throw checked exceptions
Since Java 5 Java 1.0
Usage ExecutorService.submit() ExecutorService.execute() or Thread.start()
Result Access Via Future object No direct result access

When to Use

Use Callable when you need return values or proper exception handling. Use Runnable for fire-and-forget tasks.

Practical Callable Examples

Database Query Callable

import java.util.concurrent.*;
import java.util.*;

public class DatabaseCallableExample {
    
    static class DatabaseQueryCallable implements Callable> {
        private final String query;
        private final int simulatedDelay;
        
        public DatabaseQueryCallable(String query, int delay) {
            this.query = query;
            this.simulatedDelay = delay;
        }
        
        @Override
        public List call() throws Exception {
            System.out.println("Executing query: " + query);
            
            // Simulate database query time
            Thread.sleep(simulatedDelay);
            
            // Simulate query results
            List results = Arrays.asList(
                "Result 1 for: " + query,
                "Result 2 for: " + query,
                "Result 3 for: " + query
            );
            
            // Simulate occasional database error
            if (query.contains("error")) {
                throw new SQLException("Database connection failed");
            }
            
            return results;
        }
    }
    
    public static void main(String[] args) throws Exception {
        ExecutorService executor = Executors.newFixedThreadPool(2);
        
        // Submit multiple database queries
        Future> future1 = executor.submit(
            new DatabaseQueryCallable("SELECT * FROM users", 2000));
        
        Future> future2 = executor.submit(
            new DatabaseQueryCallable("SELECT * FROM orders", 1500));
        
        // Process results as they become available
        try {
            List users = future1.get(3, TimeUnit.SECONDS);
            System.out.println("Users: " + users);
        } catch (TimeoutException e) {
            System.out.println("User query timed out");
        }
        
        List orders = future2.get();
        System.out.println("Orders: " + orders);
        
        executor.shutdown();
    }
}

File Processing Callable

import java.util.concurrent.*;
import java.nio.file.*;
import java.util.*;

public class FileProcessingCallableExample {
    
    static class FileProcessorCallable implements Callable> {
        private final String filePath;
        
        public FileProcessorCallable(String filePath) {
            this.filePath = filePath;
        }
        
        @Override
        public Map call() throws Exception {
            System.out.println("Processing file: " + filePath);
            
            // Simulate file reading and processing
            Thread.sleep(1000);
            
            // Simulate word count results
            Map wordCounts = new HashMap<>();
            wordCounts.put("java", 15);
            wordCounts.put("future", 8);
            wordCounts.put("callable", 12);
            
            // Simulate file not found error
            if (filePath.contains("missing")) {
                throw new NoSuchFileException("File not found: " + filePath);
            }
            
            return wordCounts;
        }
    }
    
    public static void main(String[] args) {
        ExecutorService executor = Executors.newCachedThreadPool();
        
        List>> futures = new ArrayList<>();
        
        // Process multiple files concurrently
        String[] files = {"document1.txt", "document2.txt", "missing_file.txt"};
        
        for (String file : files) {
            Future> future = executor.submit(
                new FileProcessorCallable(file));
            futures.add(future);
        }
        
        // Collect results
        for (int i = 0; i < futures.size(); i++) {
            try {
                Map result = futures.get(i).get();
                System.out.println("File " + files[i] + " word counts: " + result);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                System.out.println("Processing interrupted for: " + files[i]);
            } catch (ExecutionException e) {
                System.out.println("Error processing " + files[i] + ": " + 
                                 e.getCause().getMessage());
            }
        }
        
        executor.shutdown();
    }
}

public class FutureStateExample {
    public static void monitorFuture(Future future) {
        System.out.println("=== Future State ===");
        System.out.println("isDone(): " + future.isDone());
        System.out.println("isCancelled(): " + future.isCancelled());
        
        if (future.isDone() && !future.isCancelled()) {
            try {
                System.out.println("Result: " + future.get());
            } catch (Exception e) {
                System.out.println("Exception: " + e.getCause().getMessage());
            }
        }
        System.out.println("===================");
    }
}

Advanced Future Patterns

Timeout and Fallback

import java.util.concurrent.*;

public class TimeoutFallbackExample {
    
    public static String fetchDataWithTimeout(String taskName, long timeoutMs) {
        ExecutorService executor = Executors.newSingleThreadExecutor();
        
        Callable dataFetcher = () -> {
            // Simulate unpredictable processing time
            long processingTime = (long) (Math.random() * 3000);
            Thread.sleep(processingTime);
            return "Data from " + taskName + " (took " + processingTime + "ms)";
        };
        
        Future future = executor.submit(dataFetcher);
        
        try {
            // Wait with timeout
            return future.get(timeoutMs, TimeUnit.MILLISECONDS);
            
        } catch (TimeoutException e) {
            // Cancel the task if it's still running
            future.cancel(true);
            return "Fallback data for " + taskName + " (timeout after " + timeoutMs + "ms)";
            
        } catch (InterruptedException | ExecutionException e) {
            return "Error fetching data for " + taskName + ": " + e.getMessage();
            
        } finally {
            executor.shutdown();
        }
    }
    
    public static void main(String[] args) {
        String result1 = fetchDataWithTimeout("Task-A", 2000);
        String result2 = fetchDataWithTimeout("Task-B", 1000);
        
        System.out.println(result1);
        System.out.println(result2);
    }
}

Multiple Futures Coordination

import java.util.concurrent.*;
import java.util.*;

public class MultipleFuturesExample {
    
    public static void main(String[] args) throws Exception {
        ExecutorService executor = Executors.newFixedThreadPool(4);
        
        List> tasks = Arrays.asList(
            () -> { Thread.sleep(2000); return "Task-1"; },
            () -> { Thread.sleep(1000); return "Task-2"; },
            () -> { Thread.sleep(3000); return "Task-3"; },
            () -> { Thread.sleep(500); return "Task-4"; }
        );
        
        // Submit all tasks
        List> futures = new ArrayList<>();
        for (Callable task : tasks) {
            futures.add(executor.submit(task));
        }
        
        // Process completed tasks as they finish
        int completed = 0;
        while (completed < futures.size()) {
            for (Future future : futures) {
                if (future.isDone() && !future.isCancelled()) {
                    try {
                        String result = future.get();
                        System.out.println("Completed: " + result);
                        completed++;
                    } catch (ExecutionException e) {
                        System.out.println("Failed: " + e.getCause().getMessage());
                        completed++;
                    }
                }
            }
            Thread.sleep(100); // Small delay to prevent busy waiting
        }
        
        // Alternative: Wait for all with timeout
        try {
            for (Future future : futures) {
                String result = future.get(5, TimeUnit.SECONDS);
                System.out.println("Result: " + result);
            }
        } catch (TimeoutException e) {
            System.out.println("Some tasks timed out");
        }
        
        executor.shutdown();
    }
}

Future Exception Handling

import java.util.concurrent.*;
import java.util.*;

public class FutureExceptionHandling {
    
    static class FaultyCallable implements Callable {
        private final String name;
        private final boolean shouldFail;
        
        public FaultyCallable(String name, boolean shouldFail) {
            this.name = name;
            this.shouldFail = shouldFail;
        }
        
        @Override
        public String call() throws Exception {
            System.out.println("Executing: " + name);
            Thread.sleep(1000);
            
            if (shouldFail) {
                throw new RuntimeException("Simulated failure in " + name);
            }
            
            return "Success from " + name;
        }
    }
    
    public static void main(String[] args) {
        ExecutorService executor = Executors.newFixedThreadPool(2);
        
        List> futures = Arrays.asList(
            executor.submit(new FaultyCallable("Task-Good", false)),
            executor.submit(new FaultyCallable("Task-Bad", true)),
            executor.submit(new FaultyCallable("Task-Good2", false))
        );
        
        for (int i = 0; i < futures.size(); i++) {
            Future future = futures.get(i);
            try {
                // get() will throw ExecutionException if task failed
                String result = future.get();
                System.out.println("Task " + i + " succeeded: " + result);
                
            } catch (InterruptedException e) {
                System.out.println("Task " + i + " interrupted");
                Thread.currentThread().interrupt();
                
            } catch (ExecutionException e) {
                // Get the actual exception thrown by the Callable
                Throwable cause = e.getCause();
                System.out.println("Task " + i + " failed with: " + cause.getMessage());
                
                // Handle specific exception types
                if (cause instanceof RuntimeException) {
                    System.out.println("It was a RuntimeException");
                }
            }
        }
        
        executor.shutdown();
    }
}

CompletableFuture (Java 8+)

CompletableFuture Advantages

CompletableFuture is a dramatic improvement over basic Future, providing composable asynchronous programming with fluent API.

Non-blocking Composition

Chain operations without blocking threads

Exception Handling

Built-in exception recovery mechanisms

Combination

Combine multiple futures easily

Manual Completion

Complete futures manually if needed

import java.util.concurrent.*;
import java.util.*;

public class CompletableFutureBasicExample {
    
    public static void main(String[] args) throws Exception {
        // Create CompletableFuture
        CompletableFuture future = CompletableFuture.supplyAsync(() -> {
            try {
                Thread.sleep(1000);
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
            return "Hello";
        });
        
        // Chain operations
        CompletableFuture greeting = future
            .thenApply(result -> result + " World")
            .thenApply(String::toUpperCase)
            .thenApply(result -> result + "!");
        
        // Get result
        System.out.println(greeting.get()); // HELLO WORLD!
        
        // Handle exceptions
        CompletableFuture safeFuture = CompletableFuture
            .supplyAsync(() -> {
                if (Math.random() > 0.5) {
                    throw new RuntimeException("Oops!");
                }
                return "Success";
            })
            .exceptionally(ex -> "Recovered from: " + ex.getMessage());
        
        System.out.println(safeFuture.get());
    }
}

CompletableFuture Chaining

import java.util.concurrent.*;
import java.util.function.*;

public class CompletableFutureChaining {
    
    public static void main(String[] args) throws Exception {
        // thenApply - transform result
        CompletableFuture future1 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenApply(s -> s + " world")
            .thenApply(String::toUpperCase);
        
        System.out.println("thenApply: " + future1.get());
        
        // thenCompose - chain dependent futures
        CompletableFuture future2 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenCompose(s -> CompletableFuture.supplyAsync(() -> s + " world"));
        
        System.out.println("thenCompose: " + future2.get());
        
        // thenAccept - consume result without returning
        CompletableFuture future3 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenAccept(s -> System.out.println("thenAccept: " + s));
        future3.get();
        
        // thenRun - run action after completion
        CompletableFuture future4 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenRun(() -> System.out.println("thenRun: Task completed"));
        future4.get();
        
        // Handle both result and exception
        CompletableFuture future5 = CompletableFuture
            .supplyAsync(() -> {
                if (Math.random() > 0.5) throw new RuntimeException("Error");
                return "Success";
            })
            .handle((result, ex) -> {
                if (ex != null) {
                    return "Handled: " + ex.getMessage();
                }
                return "Result: " + result;
            });
        
        System.out.println("handle: " + future5.get());
    }
}

import java.util.*;

            .supplyAsync(() -> "World");
