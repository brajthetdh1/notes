Executor Framework Overview

The Java Executor Framework provides a high-level replacement for working with threads directly. It simplifies thread management and provides thread pooling capabilities.

Key Benefits

  • Simplified thread management
  • Improved performance through thread reuse
  • Better resource management
  • Built-in thread pooling
  • Task submission and execution separation

Executor Hierarchy

Executor

Base interface for executing tasks

ExecutorService

Extended interface with lifecycle methods

ThreadPoolExecutor

Most common implementation

Basic Executor Usage

import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

public class BasicExecutorExample {
    
    public static void main(String[] args) {
        // Create a thread pool with 4 threads
        ExecutorService executor = Executors.newFixedThreadPool(4);
        
        // Submit 10 tasks to the executor
        for (int i = 1; i <= 10; i++) {
            final int taskId = i;
            executor.submit(() -> {
                System.out.println("Task " + taskId + " executed by " + 
                                 Thread.currentThread().getName());
                try {
                    Thread.sleep(1000); // Simulate work
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            });
        }
        
        // Initiate orderly shutdown
        executor.shutdown();
        
        try {
            // Wait for all tasks to complete or timeout after 1 minute
            if (!executor.awaitTermination(1, TimeUnit.MINUTES)) {
                // Force shutdown if tasks don't complete in time
                executor.shutdownNow();
            }
        } catch (InterruptedException e) {
            executor.shutdownNow();
            Thread.currentThread().interrupt();
        }
        
        System.out.println("All tasks completed");
    }
}

Fixed number of threads that reuse a shared unbounded queue.

Threads: Fixed
Queue: Unbounded

Threads: Dynamic
Queue: Synchronous
Use Case: Short-lived async tasks

CachedThreadPool Example

import java.util.concurrent.*;

public class CachedThreadPoolExample {
    
    public static void main(String[] args) throws InterruptedException {
        // Create cached thread pool
        ExecutorService executor = Executors.newCachedThreadPool();
        
        System.out.println("CachedThreadPool created - will create threads as needed");
        
        // Submit tasks with different durations
        for (int i = 1; i <= 15; i++) {
            final int taskId = i;
            final long sleepTime = (i % 3 == 0) ? 4000 : 1000; // Mix of short and long tasks
            
            executor.submit(() -> {
                String threadName = Thread.currentThread().getName();
                System.out.println("Task " + taskId + " started by " + threadName + 
                                 " (sleep: " + sleepTime + "ms)");
                
                try {
                    Thread.sleep(sleepTime);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
                
                System.out.println("Task " + taskId + " completed by " + threadName);
            });
            
            // Small delay between task submissions
            Thread.sleep(100);
        }
        
        // Monitor thread creation
        Thread monitor = new Thread(() -> {
            while (!executor.isShutdown()) {
                Thread[] threads = new Thread[Thread.activeCount()];
                Thread.enumerate(threads);
                long poolThreads = java.util.Arrays.stream(threads)
                    .filter(t -> t != null && t.getName().startsWith("pool-"))
                    .count();
                System.out.println("Active pool threads: " + poolThreads);
                
                try {
                    Thread.sleep(1000);
                } catch (InterruptedException e) {
                    break;
                }
            }
        });
        monitor.setDaemon(true);
        monitor.start();
        
        // Shutdown after some time
        Thread.sleep(10000);
        executor.shutdown();
        
        boolean terminated = executor.awaitTermination(5, TimeUnit.SECONDS);
        if (!terminated) {
            executor.shutdownNow();
        }
    }
}

Scheduled Executors

ScheduledThreadPoolExecutor

Scheduled executors allow you to schedule tasks to run after a delay or periodically.

schedule()

Executes task once after specified delay

schedule(Runnable command, long delay, TimeUnit unit)

scheduleAtFixedRate()

Executes task periodically at fixed rate

scheduleAtFixedRate(Runnable command, long initialDelay, long period, TimeUnit unit)

scheduleWithFixedDelay()

Executes task periodically with fixed delay between executions

scheduleWithFixedDelay(Runnable command, long initialDelay, long delay, TimeUnit unit) 
import java.util.concurrent.*;

public class ScheduledExecutorExample {
    
    public static void main(String[] args) throws InterruptedException {
        // Create scheduled executor with 2 threads
        ScheduledExecutorService scheduler = Executors.newScheduledThreadPool(2);
        
        System.out.println("ScheduledExecutor started at: " + java.time.LocalTime.now());
        
        // Schedule a one-time task to run after 3 seconds
        ScheduledFuture oneTimeTask = scheduler.schedule(() -> {
            System.out.println("One-time task executed at: " + java.time.LocalTime.now());
        }, 3, TimeUnit.SECONDS);
        
        // Schedule a task to run every 2 seconds after initial 1 second delay
        ScheduledFuture fixedRateTask = scheduler.scheduleAtFixedRate(() -> {
            System.out.println("Fixed-rate task executed at: " + java.time.LocalTime.now());
        }, 1, 2, TimeUnit.SECONDS);
        
        // Schedule a task with fixed delay between executions
        ScheduledFuture fixedDelayTask = scheduler.scheduleWithFixedDelay(() -> {
            System.out.println("Fixed-delay task started at: " + java.time.LocalTime.now());
            try {
                Thread.sleep(1000); // Simulate work
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
            System.out.println("Fixed-delay task completed at: " + java.time.LocalTime.now());
        }, 1, 3, TimeUnit.SECONDS);
        
        // Let it run for 15 seconds
        Thread.sleep(15000);
        
        // Cancel the periodic tasks
        fixedRateTask.cancel(false);
        fixedDelayTask.cancel(false);
        
        // Shutdown the scheduler
        scheduler.shutdown();
        boolean terminated = scheduler.awaitTermination(5, TimeUnit.SECONDS);
        if (!terminated) {
            scheduler.shutdownNow();
        }
        
        System.out.println("ScheduledExecutor completed");
    }
}

Real-world: Task Scheduler with Monitoring

import java.util.concurrent.*;
import java.time.LocalTime;
import java.util.concurrent.atomic.AtomicInteger;

public class TaskSchedulerExample {
    private final ScheduledExecutorService scheduler;
    private final AtomicInteger taskCounter;
    private final AtomicInteger successCounter;
    private final AtomicInteger failureCounter;
    
    public TaskSchedulerExample() {
        this.scheduler = Executors.newScheduledThreadPool(3);
        this.taskCounter = new AtomicInteger(0);
        this.successCounter = new AtomicInteger(0);
        this.failureCounter = new AtomicInteger(0);
    }
    
    public void scheduleDatabaseCleanup() {
        // Run every hour, starting after 1 minute
        scheduler.scheduleAtFixedRate(() -> {
            int taskId = taskCounter.incrementAndGet();
            System.out.println("[" + LocalTime.now() + "] Task " + taskId + ": Starting database cleanup");
            
            try {
                // Simulate database cleanup
                Thread.sleep(5000);
                successCounter.incrementAndGet();
                System.out.println("[" + LocalTime.now() + "] Task " + taskId + ": Database cleanup completed successfully");
            } catch (InterruptedException e) {
                failureCounter.incrementAndGet();
                System.out.println("[" + LocalTime.now() + "] Task " + taskId + ": Database cleanup interrupted");
                Thread.currentThread().interrupt();
            } catch (Exception e) {
                failureCounter.incrementAndGet();
                System.out.println("[" + LocalTime.now() + "] Task " + taskId + ": Database cleanup failed: " + e.getMessage());
            }
        }, 1, 60, TimeUnit.MINUTES);
    }
    
    public void scheduleHealthCheck() {
        // Run every 30 seconds, starting after 10 seconds
        scheduler.scheduleAtFixedRate(() -> {
            System.out.println("[" + LocalTime.now() + "] Health check: System is healthy");
        }, 10, 30, TimeUnit.SECONDS);
    }
    
    public void scheduleMetricsReport() {
        // Run every 5 minutes, starting after 2 minutes
        scheduler.scheduleAtFixedRate(() -> {
            int total = taskCounter.get();
            int success = successCounter.get();
            int failure = failureCounter.get();
            
            System.out.println("=== METRICS REPORT ===");
            System.out.println("Total tasks: " + total);
            System.out.println("Successful: " + success);
            System.out.println("Failed: " + failure);
            System.out.println("Success rate: " + (total > 0 ? (success * 100.0 / total) : 0) + "%");
            System.out.println("======================");
        }, 2, 5, TimeUnit.MINUTES);
    }
    
    public void start() {
        scheduleDatabaseCleanup();
        scheduleHealthCheck();
        scheduleMetricsReport();
        System.out.println("Task scheduler started at: " + LocalTime.now());
    }
    
    public void stop() throws InterruptedException {
        System.out.println("Shutting down task scheduler...");
        scheduler.shutdown();
        
        if (scheduler.awaitTermination(10, TimeUnit.SECONDS)) {
            System.out.println("Task scheduler shutdown completed");
        } else {
            System.out.println("Forcing shutdown...");
            scheduler.shutdownNow();
        }
    }
    
    public static void main(String[] args) throws InterruptedException {
        TaskSchedulerExample scheduler = new TaskSchedulerExample();
        scheduler.start();
        
        // Let it run for 2 minutes in demo
        Thread.sleep(120000);
        
        scheduler.stop();
    }
}

ForkJoin Framework

Work-Stealing Algorithm

The ForkJoin framework is designed for recursive, divide-and-conquer algorithms that can be broken down into smaller sub-tasks.

Key Concepts

  • Fork: Split task into smaller sub-tasks
  • Join: Combine results from sub-tasks
  • Work-stealing: Idle threads steal work from busy threads
  • RecursiveTask: Returns a result
  • RecursiveAction: No result returned
import java.util.concurrent.*;

public class ForkJoinBasicExample {
    
    public static void main(String[] args) {
        // Create ForkJoinPool (uses available processors by default)
        ForkJoinPool forkJoinPool = new ForkJoinPool();
        
        // Example: Calculate sum of array using ForkJoin
        long[] numbers = new long[1000];
        for (int i = 0; i < numbers.length; i++) {
            numbers[i] = i + 1;
        }
        
        SumTask task = new SumTask(numbers, 0, numbers.length);
        Long result = forkJoinPool.invoke(task);
        
        System.out.println("Sum of 1 to 1000: " + result);
        System.out.println("Theoretical sum: " + (1000L * 1001L / 2L));
        
        forkJoinPool.shutdown();
    }
    
    // RecursiveTask for calculating sum
    static class SumTask extends RecursiveTask {
        private static final int THRESHOLD = 100; // Split if more than 100 elements
        private final long[] array;
        private final int start;
        private final int end;
        
        SumTask(long[] array, int start, int end) {
            this.array = array;
            this.start = start;
            this.end = end;
        }
        
        @Override
        protected Long compute() {
            int length = end - start;
            
            // If small enough, compute directly
            if (length <= THRESHOLD) {
                long sum = 0;
                for (int i = start; i < end; i++) {
                    sum += array[i];
                }
                return sum;
            }
            
            // Split task into two sub-tasks
            int mid = start + length / 2;
            SumTask leftTask = new SumTask(array, start, mid);
            SumTask rightTask = new SumTask(array, mid, end);
            
            // Fork the left task (execute asynchronously)
            leftTask.fork();
            
            // Compute right task and then join with left
            Long rightResult = rightTask.compute();
            Long leftResult = leftTask.join();
            
            return leftResult + rightResult;
        }
    }
}

Advanced: Parallel Array Processing

import java.util.concurrent.*;
import java.util.Arrays;
import java.util.Random;

public class ParallelArrayProcessing {
    
    public static void main(String[] args) {
        // Create large array
        int size = 10_000_000;
        int[] array = new int[size];
        Random random = new Random();
        
        // Fill with random numbers
        for (int i = 0; i < size; i++) {
            array[i] = random.nextInt(1000);
        }
        
        ForkJoinPool pool = new ForkJoinPool();
        
        // Measure sequential processing time
        long startTime = System.currentTimeMillis();
        long sequentialSum = sequentialSum(array);
        long sequentialTime = System.currentTimeMillis() - startTime;
        
        // Measure parallel processing time
        startTime = System.currentTimeMillis();
        ParallelSumTask parallelTask = new ParallelSumTask(array, 0, array.length);
        long parallelSum = pool.invoke(parallelTask);
        long parallelTime = System.currentTimeMillis() - startTime;
        
        System.out.println("Sequential sum: " + sequentialSum + " in " + sequentialTime + "ms");
        System.out.println("Parallel sum: " + parallelSum + " in " + parallelTime + "ms");
        System.out.println("Speedup: " + (sequentialTime / (double) parallelTime) + "x");
        
        pool.shutdown();
    }
    
    static long sequentialSum(int[] array) {
        long sum = 0;
        for (int value : array) {
            sum += value;
        }
        return sum;
    }
    
    static class ParallelSumTask extends RecursiveTask {
        private static final int THRESHOLD = 10_000;
        private final int[] array;
        private final int start;
        private final int end;
        
        ParallelSumTask(int[] array, int start, int end) {
            this.array = array;
            this.start = start;
            this.end = end;
        }
        
        @Override
        protected Long compute() {
            int length = end - start;
            
            if (length <= THRESHOLD) {
                // Base case: compute sequentially
                long sum = 0;
                for (int i = start; i < end; i++) {
                    sum += array[i];
                }
                return sum;
            }
            
            // Recursive case: split and conquer
            int mid = start + length / 2;
            ParallelSumTask leftTask = new ParallelSumTask(array, start, mid);
            ParallelSumTask rightTask = new ParallelSumTask(array, mid, end);
            
            leftTask.fork();
            long rightResult = rightTask.compute();
            long leftResult = leftTask.join();
            
            return leftResult + rightResult;
        }
    }
}

Best Practices

Choose Right Pool Size

For CPU-intensive tasks: number of cores. For I/O-intensive tasks: larger pools.

int cores = Runtime.getRuntime().availableProcessors();

Proper Shutdown

Always shutdown executors and handle termination properly.

executor.shutdown() + awaitTermination()

Handle Exceptions

Task exceptions are wrapped in ExecutionException - handle them properly.

future.get() catches ExecutionException

Monitor Performance

Use ThreadPoolExecutor methods to monitor pool health and performance.

getActiveCount(), getQueue().size()

Avoid CachedThreadPool

In production, avoid CachedThreadPool due to potential unbounded thread creation.

Use Callable for Results

Use Callable instead of Runnable when you need return values or exception handling.

Future future = executor.submit(callable);

Proper Executor Shutdown Pattern

import java.util.concurrent.*;

public class ProperShutdownExample {
    
    public static void shutdownExecutor(ExecutorService executor, long timeout, TimeUnit unit) {
        if (executor == null) return;
        
        // Disable new tasks from being submitted
        executor.shutdown();
        
        try {
            // Wait a while for existing tasks to terminate
            if (!executor.awaitTermination(timeout, unit)) {
                // Cancel currently executing tasks
                executor.shutdownNow();
                
                // Wait a while for tasks to respond to being cancelled
                if (!executor.awaitTermination(timeout, unit)) {
                    System.err.println("Executor did not terminate");
                }
            }
        } catch (InterruptedException ie) {
            // (Re-)Cancel if current thread also interrupted
            executor.shutdownNow();
            
            // Preserve interrupt status
            Thread.currentThread().interrupt();
        }
    }
    
    public static void main(String[] args) {
        ExecutorService executor = Executors.newFixedThreadPool(4);
        
        // Submit some tasks
        for (int i = 0; i < 10; i++) {
            final int taskId = i;
            executor.submit(() -> {
                try {
                    Thread.sleep(2000);
                    System.out.println("Task " + taskId + " completed");
                } catch (InterruptedException e) {
                    System.out.println("Task " + taskId + " interrupted");
                    Thread.currentThread().interrupt();
                }
            });
        }
        
        // Proper shutdown
        shutdownExecutor(executor, 5, TimeUnit.SECONDS);
    }
}

Custom ThreadPoolExecutor

import java.util.concurrent.*;
import java.util.concurrent.atomic.*;

public class CustomThreadPoolExample {
    
    public static void main(String[] args) {
        // Create custom ThreadPoolExecutor with monitoring
        int corePoolSize = 2;
        int maxPoolSize = 4;
        long keepAliveTime = 60L;
        TimeUnit unit = TimeUnit.SECONDS;
        BlockingQueue workQueue = new LinkedBlockingQueue<>(10);
        
        ThreadPoolExecutor executor = new ThreadPoolExecutor(
            corePoolSize, maxPoolSize, keepAliveTime, unit, workQueue,
            new CustomThreadFactory(), new CustomRejectionHandler()
        );
        
        // Add monitoring hook
        executor.setRejectedExecutionHandler(new MonitoringRejectionHandler(executor));
        
        // Submit tasks
        for (int i = 1; i <= 20; i++) {
            final int taskId = i;
            try {
                executor.submit(() -> {
                    System.out.println("Executing task " + taskId + " by " + 
                                     Thread.currentThread().getName());
                    try {
                        Thread.sleep(1000);
                    } catch (InterruptedException e) {
                        Thread.currentThread().interrupt();
                    }
                });
            } catch (RejectedExecutionException e) {
                System.out.println("Task " + taskId + " rejected: " + e.getMessage());
            }
        }
        
        // Monitor thread pool
        ScheduledExecutorService monitor = Executors.newSingleThreadScheduledExecutor();
        monitor.scheduleAtFixedRate(() -> {
            System.out.printf("Pool: %d active, %d completed, %d queued%n",
                executor.getActiveCount(),
                executor.getCompletedTaskCount(),
                executor.getQueue().size());
        }, 1, 1, TimeUnit.SECONDS);
        
        // Shutdown
        try {
            Thread.sleep(10000);
            monitor.shutdown();
            executor.shutdown();
            executor.awaitTermination(10, TimeUnit.SECONDS);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }
    
    static class CustomThreadFactory implements ThreadFactory {
        private final AtomicInteger counter = new AtomicInteger(1);
        
        @Override
        public Thread newThread(Runnable r) {
            Thread thread = new Thread(r, "CustomThread-" + counter.getAndIncrement());
            thread.setDaemon(false);
            thread.setPriority(Thread.NORM_PRIORITY);
            return thread;
        }
    }
    
    static class CustomRejectionHandler implements RejectedExecutionHandler {
        @Override
        public void rejectedExecution(Runnable r, ThreadPoolExecutor executor) {
            System.out.println("Task rejected: " + r.toString());
            throw new RejectedExecutionException("Task rejected - queue full");
        }
    }
    
    static class MonitoringRejectionHandler implements RejectedExecutionHandler {
        private final ThreadPoolExecutor executor;
        
        MonitoringRejectionHandler(ThreadPoolExecutor executor) {
            this.executor = executor;
        }
        
        @Override
        public void rejectedExecution(Runnable r, ThreadPoolExecutor executor) {
            System.err.printf("REJECTED: Active: %d, Pool: %d, Max: %d, Queue: %d%n",
                executor.getActiveCount(),
                executor.getPoolSize(),
                executor.getMaximumPoolSize(),
                executor.getQueue().size());
            throw new RejectedExecutionException("Task rejected - system overloaded");
        }
    }
}

CompletableFuture with Executors

import java.util.concurrent.*;
import java.util.*;

public class CompletableFutureExample {
    
    public static void main(String[] args) throws Exception {
        ExecutorService executor = Executors.newFixedThreadPool(4);
        
        // Basic CompletableFuture
        CompletableFuture future1 = CompletableFuture.supplyAsync(() -> {
            sleep(1000);
            return "Result from Task 1";
        }, executor);
        
        // Chain tasks
        CompletableFuture future2 = future1.thenApply(result -> {
            return result + " -> Processed by Task 2";
        });
        
        // Combine multiple futures
        CompletableFuture future3 = CompletableFuture.supplyAsync(() -> {
            sleep(500);
            return "Result from Task 3";
        }, executor);
        
        CompletableFuture future4 = CompletableFuture.supplyAsync(() -> {
            sleep(800);
            return "Result from Task 4";
        }, executor);
        
        CompletableFuture combined = future3.thenCombine(future4, (res3, res4) -> {
            return res3 + " + " + res4;
        });
        
        // Handle exceptions
        CompletableFuture future5 = CompletableFuture.supplyAsync(() -> {
            if (Math.random() > 0.5) {
                throw new RuntimeException("Simulated error");
            }
            return "Success from Task 5";
        }, executor).exceptionally(ex -> {
            return "Recovered from: " + ex.getMessage();
        });
        
        // Wait for all futures
        CompletableFuture allFutures = CompletableFuture.allOf(
            future2, combined, future5
        );
        
        // Get results
        String result2 = future2.get();
        String resultCombined = combined.get();
        String result5 = future5.get();
        
        System.out.println("Future 2: " + result2);
        System.out.println("Combined: " + resultCombined);
        System.out.println("Future 5: " + result5);
        
        executor.shutdown();
    }
    
    private static void sleep(long millis) {
        try {
            Thread.sleep(millis);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }
}

ThreadPool Monitoring & Metrics

import java.util.concurrent.*;
import java.util.concurrent.atomic.*;

public class ThreadPoolMonitor {
    private final ThreadPoolExecutor executor;
    private final AtomicLong totalTasks = new AtomicLong();
    private final AtomicLong completedTasks = new AtomicLong();
    private final AtomicLong failedTasks = new AtomicLong();
    
    public ThreadPoolMonitor(int corePoolSize, int maxPoolSize, int queueSize) {
        this.executor = new ThreadPoolExecutor(
            corePoolSize, maxPoolSize, 60L, TimeUnit.SECONDS,
            new LinkedBlockingQueue<>(queueSize),
            new MonitoringThreadFactory(),
            new MonitoringRejectionHandler()
        );
    }
    
    public void submitTask(Runnable task) {
        totalTasks.incrementAndGet();
        executor.submit(() -> {
            try {
                task.run();
                completedTasks.incrementAndGet();
            } catch (Exception e) {
                failedTasks.incrementAndGet();
                System.err.println("Task failed: " + e.getMessage());
            }
        });
    }
    
    public void printMetrics() {
        System.out.println("=== THREAD POOL METRICS ===");
        System.out.println("Active threads: " + executor.getActiveCount());
        System.out.println("Pool size: " + executor.getPoolSize());
        System.out.println("Core pool size: " + executor.getCorePoolSize());
        System.out.println("Max pool size: " + executor.getMaximumPoolSize());
        System.out.println("Queue size: " + executor.getQueue().size());
        System.out.println("Completed tasks: " + executor.getCompletedTaskCount());
        System.out.println("Total submitted: " + totalTasks.get());
        System.out.println("Successfully completed: " + completedTasks.get());
        System.out.println("Failed: " + failedTasks.get());
        System.out.println("===========================");
    }
    
    public void shutdown() throws InterruptedException {
        executor.shutdown();
        if (!executor.awaitTermination(10, TimeUnit.SECONDS)) {
            executor.shutdownNow();
        }
    }
    
    static class MonitoringThreadFactory implements ThreadFactory {
        private final AtomicInteger counter = new AtomicInteger(1);
        
        @Override
        public Thread newThread(Runnable r) {
            Thread thread = new Thread(r, "MonitorThread-" + counter.getAndIncrement());
            thread.setUncaughtExceptionHandler((t, e) -> {
                System.err.println("Uncaught exception in thread " + t.getName() + ": " + e.getMessage());
            });
            return thread;
        }
    }
    
    static class MonitoringRejectionHandler implements RejectedExecutionHandler {
        @Override
        public void rejectedExecution(Runnable r, ThreadPoolExecutor executor) {
            System.err.println("Task rejected - thread pool overloaded");
        }
    }
    
    public static void main(String[] args) throws InterruptedException {
        ThreadPoolMonitor monitor = new ThreadPoolMonitor(2, 4, 5);
        
        // Submit tasks
        for (int i = 0; i < 20; i++) {
            final int taskId = i;
            monitor.submitTask(() -> {
                System.out.println("Executing task " + taskId);
                try {
                    Thread.sleep(1000);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            });
            
            // Print metrics every 3 tasks
            if (i % 3 == 0) {
                monitor.printMetrics();
            }
            
            Thread.sleep(100);
        }
        
        // Final metrics
        Thread.sleep(2000);
        monitor.printMetrics();
        monitor.shutdown();
    }
