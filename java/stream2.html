// Person class used throughout examples
public class Person {
    private String name;
    private int age;
    private String department;
    private double salary;
    private String city;
    
    // Constructor
    public Person(String name, int age, String department, double salary, String city) {
        this.name = name;
        this.age = age;
        this.department = department;
        this.salary = salary;
        this.city = city;
    }
    
    // Getters and setters
    public String getName() { return name; }
    public int getAge() { return age; }
    public String getDepartment() { return department; }
    public double getSalary() { return salary; }
    public String getCity() { return city; }
    
    @Override
    public String toString() {
        return String.format("Person{name='%s', age=%d, dept='%s', salary=%.2f, city='%s'}",
                name, age, department, salary, city);
    }
    
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Person)) return false;
        Person person = (Person) o;
        return age == person.age && 
               Double.compare(person.salary, salary) == 0 &&
               Objects.equals(name, person.name) &&
               Objects.equals(department, person.department) &&
               Objects.equals(city, person.city);
    }
    
    @Override
    public int hashCode() {
        return Objects.hash(name, age, department, salary, city);
    }
}

/**
 * Processes a list of people and prints detailed information
 * Demonstrates multiple operations in a single pipeline
 */
public void processAndPrintPeople(List<Person> people) {
    System.out.println("=== Processing People Data ===");
    
    people.stream()
        .filter(person -> person.getAge() > 25)           // Filter adults
        .sorted(Comparator.comparing(Person::getName))    // Sort by name
        .map(person -> String.format(                     // Transform to formatted string
            "Name: %-12s | Age: %2d | Dept: %-12s | Salary: $%,8.2f",
            person.getName(), 
            person.getAge(), 
            person.getDepartment(), 
            person.getSalary()
        ))
        .forEach(System.out::println);                    // Print each line
    
    System.out.println("=== Processing Complete ===\n");
}

/**
 * Demonstrates complex data processing with multiple transformations
 */
public void comprehensiveDataAnalysis(List<Person> people) {
    System.out.println("=== Comprehensive Data Analysis ===");
    
    // Multiple processing steps
    List<String> processedData = people.stream()
        .peek(p -> System.out.println("Processing: " + p.getName())) // Debug
        .filter(p -> p.getSalary() > 60000)              // Filter by salary
        .sorted(Comparator
            .comparing(Person::getDepartment)            // Sort by department
            .thenComparing(Person::getSalary).reversed() // Then by salary desc
        )
        .map(p -> {                                     // Complex mapping
            String performance = p.getSalary() > 80000 ? "High" : "Medium";
            return String.format("%s (%s) - %s - $%,.0f", 
                p.getName(), p.getDepartment(), performance, p.getSalary());
        })
        .collect(Collectors.toList());                  // Collect results
    
    // Print results
    processedData.forEach(System.out::println);
    
    // Additional statistics
    long highPerformers = processedData.stream()
        .filter(s -> s.contains("High"))
        .count();
    
    System.out.printf("\nHigh performers count: %d%n", highPerformers);
    System.out.println("=== Analysis Complete ===\n");
}

/**
 * Validates and cleans person data with multiple checks
 */
public List<Person> validateAndCleanData(List<Person> people) {
    System.out.println("=== Data Validation & Cleaning ===");
    
    List<Person> cleanData = people.stream()
        .filter(Objects::nonNull)                        // Remove nulls
        .filter(p -> p.getName() != null && !p.getName().trim().isEmpty())
        .filter(p -> p.getAge() >= 18 && p.getAge() <= 100) // Validate age
        .filter(p -> p.getSalary() > 0)                  // Validate salary
        .filter(p -> p.getDepartment() != null && !p.getDepartment().isEmpty())
        .map(p -> {                                      // Clean data
            String cleanName = p.getName().trim();
            String cleanDept = p.getDepartment().trim().toUpperCase();
            return new Person(cleanName, p.getAge(), cleanDept, 
                            p.getSalary(), p.getCity());
        })
        .distinct()                                      // Remove duplicates
        .collect(Collectors.toList());
    
    long removedCount = people.size() - cleanData.size();
    System.out.printf("Removed %d invalid records%n", removedCount);
    System.out.printf("Clean records: %d%n", cleanData.size());
    System.out.println("=== Validation Complete ===\n");
    
    return cleanData;
}

}

/**
 * Filters people based on multiple complex criteria
 */
public List<Person> filterByMultipleCriteria(List<Person> people) {
    System.out.println("=== Multi-Criteria Filtering ===");
    
    Predicate<Person> ageFilter = p -> p.getAge() >= 25 && p.getAge() <= 40;
    Predicate<Person> salaryFilter = p -> p.getSalary() >= 60000;
    Predicate<Person> departmentFilter = p -> 
        "Engineering".equals(p.getDepartment()) || 
        "Marketing".equals(p.getDepartment());
    Predicate<Person> cityFilter = p -> 
        !"Boston".equals(p.getCity()); // Exclude Boston
    
    List<Person> filtered = people.stream()
        .filter(ageFilter)
        .filter(salaryFilter)
        .filter(departmentFilter)
        .filter(cityFilter)
        .sorted(Comparator.comparing(Person::getSalary).reversed())
        .collect(Collectors.toList());
    
    System.out.printf("Found %d people matching all criteria:%n", filtered.size());
    filtered.forEach(p -> System.out.println("  - " + p.getName()));
    System.out.println("=== Filtering Complete ===\n");
    
    return filtered;
}

/**
 * Dynamic filtering based on user input
 */
public List<Person> dynamicFiltering(List<Person> people, 
                                   int minAge, int maxAge, 
                                   double minSalary, List<String> departments) {
    System.out.println("=== Dynamic Filtering ===");
    
    List<Person> result = people.stream()
        .filter(p -> p.getAge() >= minAge && p.getAge() <= maxAge)
        .filter(p -> p.getSalary() >= minSalary)
        .filter(p -> departments.contains(p.getDepartment()))
        .collect(Collectors.toList());
    
    System.out.printf("Dynamic filter results (%d-%d years, $%,.0f+ salary, depts: %s):%n",
                     minAge, maxAge, minSalary, departments);
    result.forEach(p -> System.out.printf("  - %s (%d, $%,.0f, %s)%n",
        p.getName(), p.getAge(), p.getSalary(), p.getDepartment()));
    
    return result;
}

/**
 * Conditional filtering with different strategies
 */
public void conditionalFilteringExamples(List<Person> people) {
    System.out.println("=== Conditional Filtering ===");
    
    // Filter based on external conditions
    boolean includeHighSalary = true;
    boolean excludeYoungEmployees = false;
    
    List<Person> conditionalResult = people.stream()
        .filter(p -> !excludeYoungEmployees || p.getAge() >= 25)
        .filter(p -> !includeHighSalary || p.getSalary() > 70000)
        .collect(Collectors.toList());
    
    System.out.println("Conditional filtering results:");
    conditionalResult.forEach(p -> System.out.println("  - " + p.getName()));
    
    // Chain filters with different conditions
    System.out.println("\n=== Chained Filtering ===");
    people.stream()
        .filter(p -> p.getDepartment().equals("Engineering"))
        .filter(p -> p.getSalary() > 75000)
        .filter(p -> p.getAge() < 35)
        .findFirst()
        .ifPresentOrElse(
            p -> System.out.println("Found young, high-paid engineer: " + p.getName()),
            () -> System.out.println("No matching engineers found")
        );
}

                person.getName(),
                person.getDepartment(),
                experienceLevel,
                person.getSalary(),
                bonus,
                totalCompensation
            );
        })
        .sorted()
        .collect(Collectors.toList());
    
    System.out.println("Transformed Employee Data:");
    System.out.println("Name         | Department | Level    | Salary     | Bonus  | Total");
    System.out.println("-".repeat(80));
    transformedData.forEach(System.out::println);
    System.out.println("=== Transformation Complete ===\n");

/**
 * Demonstrates flatMap for processing nested collections
 */
public void processNestedCollections() {
    System.out.println("=== Nested Collection Processing ===");
    
    List<List<Integer>> nestedNumbers = Arrays.asList(
        Arrays.asList(1, 2, 3),
        Arrays.asList(4, 5, 6),
        Arrays.asList(7, 8, 9, 10)
    );
    
    // Flatten and process
    List<Integer> processedNumbers = nestedNumbers.stream()
        .flatMap(List::stream)                    // Flatten nested lists
        .map(number -> number * 2)                // Transform each number
        .filter(number -> number > 5)             // Filter results
        .sorted(Comparator.reverseOrder())        // Sort descending
        .collect(Collectors.toList());
    
    System.out.println("Processed numbers: " + processedNumbers);
    
    // More complex flatMap example
    List<String> sentences = Arrays.asList(
        "Hello world",
        "Java streams are powerful",
        "Functional programming is fun"
    );
    
    List<String> uniqueWords = sentences.stream()
        .flatMap(sentence -> Arrays.stream(sentence.split(" "))) // Split into words
        .map(String::toLowerCase)                 // Normalize case
        .distinct()                               // Remove duplicates
        .sorted()                                 // Alphabetical order
        .collect(Collectors.toList());
    
    System.out.println("Unique words: " + uniqueWords);
    System.out.println("=== Nested Processing Complete ===\n");
}

/**
 * Complex transformation with multiple steps
 */
public void multiStepTransformation(List<Person> people) {
    System.out.println("=== Multi-Step Transformation ===");
    
    Map<String, Double> departmentStats = people.stream()
        .collect(Collectors.groupingBy(
            Person::getDepartment,
            Collectors.averagingDouble(Person::getSalary)
        ));
    
    List<String> analysis = people.stream()
        .map(person -> {
            double departmentAvg = departmentStats.get(person.getDepartment());
            double difference = person.getSalary() - departmentAvg;
            String comparison = difference > 0 ? "above" : "below";
            
            return String.format(
                "%-12s: $%,8.0f (%s avg by $%,.0f)",
                person.getName(),
                person.getSalary(),
                comparison,
                Math.abs(difference)
            );
        })
        .sorted()
        .collect(Collectors.toList());
    
    System.out.println("Salary Comparison to Department Average:");
    analysis.forEach(System.out::println);
    System.out.println("=== Multi-Step Complete ===\n");
}

/**
 * Custom reduction operations for complex aggregations
 */
public void customReductionOperations(List<Person> people) {
    System.out.println("=== Custom Reduction Operations ===");
    
    // Custom aggregation: concatenate names with formatting
    String allNames = people.stream()
        .map(Person::getName)
        .sorted()
        .reduce("Employees: ", (partial, name) -> partial + name + ", ");
    
    // Remove trailing comma and space
    if (allNames.length() > 12) {
        allNames = allNames.substring(0, allNames.length() - 2);
    }
    System.out.println(allNames);
    
    // Complex reduction: calculate weighted average salary by age
    double weightedAvgSalary = people.stream()
        .reduce(new double[2], // [totalWeightedSalary, totalWeight]
            (acc, person) -> {
                double weight = person.getAge(); // Use age as weight
                acc[0] += person.getSalary() * weight;
                acc[1] += weight;
                return acc;
            },
            (acc1, acc2) -> {
                acc1[0] += acc2[0];
                acc1[1] += acc2[1];
                return acc1;
            })[0]; // Get weighted sum
    
    double totalWeight = people.stream()
        .mapToDouble(Person::getAge)
        .sum();
    
    if (totalWeight > 0) {
        weightedAvgSalary /= totalWeight;
        System.out.printf("Weighted average salary (by age): $%,.0f%n", weightedAvgSalary);
    }
    
    // Find extremes with custom comparators
    Optional<Person> highestPaidYoung = people.stream()
        .filter(p -> p.getAge() < 30)
        .max(Comparator.comparingDouble(Person::getSalary));
    
    Optional<Person> lowestPaidExperienced = people.stream()
        .filter(p -> p.getAge() >= 35)
        .min(Comparator.comparingDouble(Person::getSalary));
    
    highestPaidYoung.ifPresent(p -> 
        System.out.printf("Highest paid under 30: %s ($%,.0f)%n", 
                         p.getName(), p.getSalary()));
    
    lowestPaidExperienced.ifPresent(p -> 
        System.out.printf("Lowest paid 35+: %s ($%,.0f)%n", 
                         p.getName(), p.getSalary()));
    
    System.out.println("=== Custom Reductions Complete ===\n");
}

/**
 * Product inventory analysis with multiple aggregations
 */
public void productInventoryAnalysis(List<Product> products) {
    System.out.println("=== Product Inventory Analysis ===");
    
    // Total inventory value
    double totalInventoryValue = products.stream()
        .mapToDouble(Product::getTotalValue)
        .sum();
    
    // Value by category
    Map<String, Double> valueByCategory = products.stream()
        .collect(Collectors.groupingBy(
            Product::getCategory,
            Collectors.summingDouble(Product::getTotalValue)
        ));
    
    // High-value products (top 3 by total value)
    List<String> highValueProducts = products.stream()
        .sorted(Comparator.comparingDouble(Product::getTotalValue).reversed())
        .limit(3)
        .map(p -> String.format("%s ($%,.0f)", p.getName(), p.getTotalValue()))
        .collect(Collectors.toList());
    
    // Low-stock alert
    List<String> lowStockProducts = products.stream()
        .filter(p -> p.getQuantity() < 15)
        .sorted(Comparator.comparingInt(Product::getQuantity))
        .map(p -> String.format("%s (%d left)", p.getName(), p.getQuantity()))
        .collect(Collectors.toList());
    
    System.out.printf("Total Inventory Value: $%,.2f%n", totalInventoryValue);
    
    System.out.println("\nValue by Category:");
    valueByCategory.forEach((category, value) -> 
        System.out.printf("  %-12s: $%,.2f%n", category, value));
    
    System.out.println("\nHigh-Value Products:");
    highValueProducts.forEach(product -> System.out.println("  • " + product));
    
    if (!lowStockProducts.isEmpty()) {
        System.out.println("\n⚠️  Low Stock Alert:");
        lowStockProducts.forEach(product -> System.out.println("  • " + product));
    }
    
    System.out.println("=== Inventory Analysis Complete ===\n");
}
