CompletableFuture Overview

CompletableFuture is a dramatic enhancement to Java's Future interface, providing a powerful framework for asynchronous, non-blocking programming with fluent API for composing, combining, and executing asynchronous computation steps.

Key Advantages

  • Non-blocking operation chaining
  • Built-in exception handling
  • Manual completion control
  • Multiple future combination
  • Fluent, expressive API
  • Timeout support

CompletableFuture vs Future

Future (Basic)

• Blocking get() method
• No composition
• Manual cancellation

CompletableFuture

• Non-blocking chains
• Rich composition
• Built-in timeouts

Basic CompletableFuture Example

import java.util.concurrent.*;
import java.util.*;

public class BasicCompletableFutureExample {
    
    public static void main(String[] args) throws Exception {
        // Create a CompletableFuture that runs asynchronously
        CompletableFuture<String> future = CompletableFuture.supplyAsync(() -> {
            // Simulate long-running computation
            try {
                Thread.sleep(2000);
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
            return "Hello, World!";
        });
        
        System.out.println("Task submitted, doing other work...");
        
        // Do other work while the future is computing
        Thread.sleep(1000);
        System.out.println("Other work completed");
        
        // Get the result (blocks until available)
        String result = future.get();
        System.out.println("Result: " + result);
        
        // Non-blocking approach with callback
        CompletableFuture.supplyAsync(() -> "Async Result")
            .thenApply(String::toUpperCase)
            .thenAccept(result -> System.out.println("Callback: " + result))
            .join(); // Wait for completion
    }
}

Factory Methods

Runs Supplier asynchronously and returns CompletableFuture with result

runAsync()

Runs Runnable asynchronously and returns CompletableFuture<Void>

Returns already completed CompletableFuture with given value

import java.util.concurrent.*;

public class CreationMethods {
    
    public static void main(String[] args) throws Exception {
        // 1. supplyAsync - with result
        CompletableFuture<String> future1 = CompletableFuture.supplyAsync(() -> {
            sleep(1000);
            return "Result from supplyAsync";
        });
        
        // 2. runAsync - no result
        CompletableFuture<Void> future2 = CompletableFuture.runAsync(() -> {
            sleep(500);
            System.out.println("Task completed in runAsync");
        });
        
        // 3. completedFuture - immediately completed
        CompletableFuture<String> future3 = CompletableFuture.completedFuture("Immediate result");
        
        // 4. With custom executor
        ExecutorService customExecutor = Executors.newFixedThreadPool(2);
        CompletableFuture<String> future4 = CompletableFuture.supplyAsync(() -> {
            sleep(800);
            return "Result with custom executor";
        }, customExecutor);
        
        System.out.println("Future1: " + future1.get());
        future2.get(); // Wait for completion
        System.out.println("Future3: " + future3.get());
        System.out.println("Future4: " + future4.get());
        
        customExecutor.shutdown();
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

Manual Completion

complete()

Manually complete the future with a value

boolean complete(T value)

completeExceptionally()

Manually complete the future with an exception

boolean completeExceptionally(Throwable ex)

cancel()

Cancel the future (similar to completeExceptionally)

boolean cancel(boolean mayInterruptIfRunning) 
import java.util.concurrent.*;

public class ManualCompletion {
    
    public static void main(String[] args) throws Exception {
        // Create an incomplete future
        CompletableFuture<String> future = new CompletableFuture<>();
        
        // Start a thread that might complete the future
        new Thread(() -> {
            try {
                Thread.sleep(1000);
                
                // Simulate condition - complete manually
                if (Math.random() > 0.5) {
                    future.complete("Successfully completed");
                } else {
                    future.completeExceptionally(new RuntimeException("Simulated error"));
                }
            } catch (InterruptedException e) {
                future.completeExceptionally(e);
            }
        }).start();
        
        // Another thread can also complete it
        new Thread(() -> {
            try {
                Thread.sleep(500);
                // Force completion if not already done
                if (!future.isDone()) {
                    future.complete("Forced completion");
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }).start();
        
        // Handle the result
        try {
            String result = future.get(2, TimeUnit.SECONDS);
            System.out.println("Result: " + result);
        } catch (ExecutionException e) {
            System.out.println("Failed: " + e.getCause().getMessage());
        } catch (TimeoutException e) {
            System.out.println("Timeout - canceling");
            future.cancel(true);
        }
        
        System.out.println("Final state - Done: " + future.isDone() + 
                         ", Cancelled: " + future.isCancelled() +
                         ", Completed: " + future.isCompletedExceptionally());
    }
}

Advanced Creation Patterns

Timeout Creation Pattern

import java.util.concurrent.*;

public class TimeoutCreation {
    
    public static <T> CompletableFuture<T> withTimeout(
            Supplier<T> supplier, long timeout, TimeUnit unit) {
        
        CompletableFuture<T> future = CompletableFuture.supplyAsync(supplier);
        
        // Create timeout future
        CompletableFuture<T> timeoutFuture = CompletableFuture.supplyAsync(() -> {
            try {
                Thread.sleep(unit.toMillis(timeout));
                throw new TimeoutException("Operation timed out after " + timeout + " " + unit);
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
        });
        
        // Return whichever completes first
        return future.applyToEither(timeoutFuture, Function.identity())
                    .exceptionally(throwable -> {
                        // Cancel the original future on timeout
                        if (throwable instanceof TimeoutException) {
                            future.cancel(true);
                        }
                        throw new CompletionException(throwable);
                    });
    }
    
    public static void main(String[] args) {
        CompletableFuture<String> future = withTimeout(() -> {
            sleep(3000); // This will take 3 seconds
            return "Slow result";
        }, 2, TimeUnit.SECONDS); // Timeout after 2 seconds
        
        try {
            String result = future.get();
            System.out.println("Result: " + result);
        } catch (Exception e) {
            System.out.println("Error: " + e.getCause().getMessage());
        }
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

Retry Creation Pattern

import java.util.concurrent.*;
import java.util.function.*;
import java.util.concurrent.atomic.*;

public class RetryCreation {
    
    public static <T> CompletableFuture<T> retryAsync(
            Supplier<CompletableFuture<T>> task, 
            int maxRetries,
            Function<Integer, Long> delayCalculator) {
        
        return task.get().handle((result, throwable) -> {
            if (throwable == null) {
                return CompletableFuture.completedFuture(result);
            } else if (maxRetries > 0) {
                long delay = delayCalculator.apply(maxRetries);
                System.out.println("Retry " + maxRetries + " after " + delay + "ms");
                
                return CompletableFuture
                    .supplyAsync(() -> null)
                    .thenCompose(v -> {
                        try {
                            Thread.sleep(delay);
                        } catch (InterruptedException e) {
                            Thread.currentThread().interrupt();
                            return CompletableFuture.failedFuture(e);
                        }
                        return retryAsync(task, maxRetries - 1, delayCalculator);
                    });
            } else {
                return CompletableFuture.failedFuture(throwable);
            }
        }).thenCompose(Function.identity());
    }
    
    public static void main(String[] args) throws Exception {
        AtomicInteger attempts = new AtomicInteger(0);
        
        Supplier<CompletableFuture<String>> flakyTask = () -> 
            CompletableFuture.supplyAsync(() -> {
                int attempt = attempts.incrementAndGet();
                if (attempt < 3) {
                    throw new RuntimeException("Failed on attempt " + attempt);
                }
                return "Success on attempt " + attempt;
            });
        
        // Exponential backoff: 100ms, 200ms, 400ms...
        Function<Integer, Long> backoff = retries -> (long) (100 * Math.pow(2, 3 - retries));
        
        CompletableFuture<String> result = retryAsync(flakyTask, 3, backoff);
        System.out.println("Final: " + result.get());
    }
}

Chaining Operations

Transformation Methods

Chaining Flow

supplyAsync() → Start
thenApply() → Transform
thenCompose() → Chain Async
thenAccept() → Consume

thenApply()

Transforms result synchronously

thenApply(Function<T, U> fn)

thenCompose()

Chains another CompletableFuture

thenCompose(Function<T, CompletableFuture<U>> fn)

thenAccept()

Consumes result without returning

thenAccept(Consumer<T> action) 
import java.util.concurrent.*;
import java.util.function.*;

public class ChainingMethods {
    
    public static void main(String[] args) throws Exception {
        // thenApply - synchronous transformation
        CompletableFuture<String> future1 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenApply(s -> s + " world")
            .thenApply(String::toUpperCase);
        
        System.out.println("thenApply: " + future1.get());
        
        // thenCompose - chain dependent async operations
        CompletableFuture<String> future2 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenCompose(s -> CompletableFuture.supplyAsync(() -> s + " world"))
            .thenCompose(s -> CompletableFuture.supplyAsync(() -> s + "!"));
        
        System.out.println("thenCompose: " + future2.get());
        
        // thenAccept - consume result
        CompletableFuture<Void> future3 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenApply(String::toUpperCase)
            .thenAccept(s -> System.out.println("thenAccept: " + s));
        
        future3.get(); // Wait for completion
        
        // thenRun - run action after completion
        CompletableFuture<Void> future4 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenRun(() -> System.out.println("thenRun: Task completed"));
        
        future4.get();
        
        // Async variants - execute on different threads
        CompletableFuture<String> future5 = CompletableFuture
            .supplyAsync(() -> "hello")
            .thenApplyAsync(s -> s + " async")
            .thenApplyAsync(String::toUpperCase);
        
        System.out.println("Async chain: " + future5.get());
    }
    
    // Real-world chaining example
    public static CompletableFuture<String> processUserData(int userId) {
        return CompletableFuture
            .supplyAsync(() -> fetchUser(userId))
            .thenApply(user -> validateUser(user))
            .thenApply(user -> enrichUserData(user))
            .thenApply(user -> convertToJSON(user));
    }
    
    private static String fetchUser(int id) { 
        sleep(100); return "User" + id; 
    }
    private static String validateUser(String user) { 
        return user + "-validated"; 
    }
    private static String enrichUserData(String user) { 
        return user + "-enriched"; 
    }
    private static String convertToJSON(String user) { 
        return "{\"user\": \"" + user + "\"}"; 
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

Combining Futures

Binary Combination

thenCombine()

Combine results of two independent futures

thenCombine(CompletionStage, BiFunction)

thenAcceptBoth()

Consume results of two futures

thenAcceptBoth(CompletionStage, BiConsumer)

runAfterBoth()

Run after both complete

runAfterBoth(CompletionStage, Runnable) 
import java.util.concurrent.*;

public class BinaryCombination {
    
    public static void main(String[] args) throws Exception {
        CompletableFuture<String> future1 = CompletableFuture
            .supplyAsync(() -> { sleep(1000); return "Hello"; });
        
        CompletableFuture<String> future2 = CompletableFuture
            .supplyAsync(() -> { sleep(500); return "World"; });
        
        // thenCombine - combine results
        CompletableFuture<String> combined = future1
            .thenCombine(future2, (s1, s2) -> s1 + " " + s2 + "!");
        System.out.println("Combined: " + combined.get());
        
        // thenAcceptBoth - consume both results
        CompletableFuture<Void> consumed = future1
            .thenAcceptBoth(future2, (s1, s2) -> 
                System.out.println("Consumed: " + s1 + " and " + s2));
        consumed.get();
        
        // runAfterBoth - run after both complete
        CompletableFuture<Void> afterBoth = future1
            .runAfterBoth(future2, () -> 
                System.out.println("Both tasks completed"));
        afterBoth.get();
        
        // applyToEither - use result from whichever completes first
        CompletableFuture<String> first = future1
            .applyToEither(future2, result -> "First: " + result);
        System.out.println("First completed: " + first.get());
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

Multiple Combination

allOf()

Wait for all futures to complete

CompletableFuture.allOf(CompletableFuture...)

anyOf()

Complete when any future completes

CompletableFuture.anyOf(CompletableFuture...) 
import java.util.concurrent.*;
import java.util.*;
import java.util.stream.*;

public class MultipleCombination {
    
    public static void main(String[] args) throws Exception {
        // Create multiple futures with different completion times
        CompletableFuture<String>[] futures = new CompletableFuture[] {
            CompletableFuture.supplyAsync(() -> { sleep(2000); return "Task-A"; }),
            CompletableFuture.supplyAsync(() -> { sleep(1000); return "Task-B"; }),
            CompletableFuture.supplyAsync(() -> { sleep(1500); return "Task-C"; })
        };
        
        // allOf - wait for all to complete
        CompletableFuture<Void> allFutures = CompletableFuture.allOf(futures);
        
        // Process results when all complete
        CompletableFuture<List<String>> allResults = allFutures
            .thenApply(v -> Arrays.stream(futures)
                .map(CompletableFuture::join)
                .collect(Collectors.toList()));
        
        System.out.println("All results: " + allResults.get());
        
        // anyOf - get first completed
        CompletableFuture<Object> firstCompleted = CompletableFuture.anyOf(futures);
        System.out.println("First completed: " + firstCompleted.get());
        
        // Process multiple futures with results
        List<CompletableFuture<String>> futureList = Arrays.asList(
            CompletableFuture.supplyAsync(() -> { sleep(800); return "Data-1"; }),
            CompletableFuture.supplyAsync(() -> { sleep(1200); return "Data-2"; }),
            CompletableFuture.supplyAsync(() -> { sleep(600); return "Data-3"; })
        );
        
        // Convert List<CompletableFuture> to CompletableFuture<List>
        CompletableFuture<List<String>> combinedList = 
            CompletableFuture.allOf(futureList.toArray(new CompletableFuture[0]))
                .thenApply(v -> futureList.stream()
                    .map(CompletableFuture::join)
                    .collect(Collectors.toList()));
        
        System.out.println("Combined list: " + combinedList.get());
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

Real-world: Service Aggregation

import java.util.concurrent.*;
import java.util.*;

public class ServiceAggregation {
    
    static class UserService {
        CompletableFuture<String> fetchUser(int userId) {
            return CompletableFuture.supplyAsync(() -> {
                sleep(1000);
                return "User-" + userId;
            });
        }
    }
    
    static class OrderService {
        CompletableFuture<List<String>> fetchOrders(int userId) {
            return CompletableFuture.supplyAsync(() -> {
                sleep(800);
                return Arrays.asList("Order-1", "Order-2");
            });
        }
    }
    
    static class PaymentService {
        CompletableFuture<Double> fetchBalance(int userId) {
            return CompletableFuture.supplyAsync(() -> {
                sleep(600);
                return 1500.75;
            });
        }
    }
    
    static class RecommendationService {
        CompletableFuture<List<String>> fetchRecommendations(int userId) {
            return CompletableFuture.supplyAsync(() -> {
                sleep(1200);
                return Arrays.asList("Product-A", "Product-B", "Product-C");
            });
        }
    }
    
    public static void main(String[] args) throws Exception {
        UserService userService = new UserService();
        OrderService orderService = new OrderService();
        PaymentService paymentService = new PaymentService();
        RecommendationService recommendationService = new RecommendationService();
        
        int userId = 123;
        
        // Fetch all data concurrently
        CompletableFuture<String> userFuture = userService.fetchUser(userId);
        CompletableFuture<List<String>> ordersFuture = orderService.fetchOrders(userId);
        CompletableFuture<Double> balanceFuture = paymentService.fetchBalance(userId);
        CompletableFuture<List<String>> recommendationsFuture = 
            recommendationService.fetchRecommendations(userId);
        
        // Combine all results
        CompletableFuture<Void> allData = CompletableFuture.allOf(
            userFuture, ordersFuture, balanceFuture, recommendationsFuture);
        
        // Process combined results
        CompletableFuture<String> result = allData.thenApply(v -> {
            String user = userFuture.join();
            List<String> orders = ordersFuture.join();
            Double balance = balanceFuture.join();
            List<String> recommendations = recommendationsFuture.join();
            
            return String.format(
                "User: %s, Orders: %s, Balance: $%.2f, Recommendations: %s",
                user, orders, balance, recommendations);
        });
        
        System.out.println("Aggregated result: " + result.get());
        
        // Alternative: Combine specific pairs
        CompletableFuture<String> userWithOrders = userFuture
            .thenCombine(ordersFuture, (user, orders) -> 
                user + " has orders: " + orders);
        
        CompletableFuture<String> userWithBalance = userFuture
            .thenCombine(balanceFuture, (user, balance) -> 
                user + " has balance: $" + balance);
        
        System.out.println("User with orders: " + userWithOrders.get());
        System.out.println("User with balance: " + userWithBalance.get());
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

Exception Handling

Basic Exception Handling

exceptionally()

Provide fallback value when exception occurs

exceptionally(Function<Throwable, T> fn)

handle()

Handle both success and failure cases

handle(BiFunction<T, Throwable, U> fn)

whenComplete()

Execute action regardless of success/failure

whenComplete(BiConsumer<T, Throwable> action) 
import java.util.concurrent.*;

public class BasicExceptionHandling {
    
    public static void main(String[] args) throws Exception {
        // exceptionally - provide fallback
        CompletableFuture<String> future1 = CompletableFuture
            .supplyAsync(() -> {
                if (Math.random() > 0.5) {
                    throw new RuntimeException("Random failure");
                }
                return "Success";
            })
            .exceptionally(throwable -> "Fallback: " + throwable.getMessage());
        
        System.out.println("Future1: " + future1.get());
        
        // handle - process both success and failure
        CompletableFuture<String> future2 = CompletableFuture
            .supplyAsync(() -> {
                if (Math.random() > 0.5) {
                    throw new RuntimeException("Another failure");
                }
                return "Success";
            })
            .handle((result, throwable) -> {
                if (throwable != null) {
                    return "Handled: " + throwable.getMessage();
                }
                return "Result: " + result;
            });
        
        System.out.println("Future2: " + future2.get());
        
        // whenComplete - execute action regardless of outcome
        CompletableFuture<String> future3 = CompletableFuture
            .supplyAsync(() -> {
                if (Math.random() > 0.5) {
                    throw new RuntimeException("Final failure");
                }
                return "Success";
            })
            .whenComplete((result, throwable) -> {
                if (throwable != null) {
                    System.out.println("Completed with error: " + throwable.getMessage());
                } else {
                    System.out.println("Completed successfully: " + result);
                }
            });
        
        try {
            System.out.println("Future3: " + future3.get());
        } catch (Exception e) {
            System.out.println("Future3 failed: " + e.getCause().getMessage());
        }
    }
}

Advanced Exception Patterns

import java.util.concurrent.*;
import java.util.function.*;

public class AdvancedExceptionHandling {
    
    // Retry with exponential backoff and exception handling
    public static <T> CompletableFuture<T> retryWithBackoff(
            Supplier<CompletableFuture<T>> task, 
            int maxRetries,
            Function<Throwable, Boolean> shouldRetry) {
        
        return task.get().handle((result, throwable) -> {
            if (throwable == null) {
                return CompletableFuture.completedFuture(result);
            } else if (maxRetries > 0 && shouldRetry.apply(throwable)) {
                System.out.println("Retrying (" + maxRetries + " attempts left)...");
                
                return CompletableFuture
                    .supplyAsync(() -> null)
                    .thenCompose(v -> {
                        try {
                            long delay = (long) (100 * Math.pow(2, 3 - maxRetries));
                            Thread.sleep(delay);
                        } catch (InterruptedException e) {
                            return CompletableFuture.failedFuture(e);
                        }
                        return retryWithBackoff(task, maxRetries - 1, shouldRetry);
                    });
            } else {
                return CompletableFuture.failedFuture(throwable);
            }
        }).thenCompose(Function.identity());
    }
    
    public static void main(String[] args) throws Exception {
        // Test retry mechanism
        java.util.concurrent.atomic.AtomicInteger attempts = new java.util.concurrent.atomic.AtomicInteger();
        
        Supplier<CompletableFuture<String>> flakyTask = () -> 
            CompletableFuture.supplyAsync(() -> {
                int attempt = attempts.incrementAndGet();
                if (attempt < 3) {
                    throw new RuntimeException("Temporary failure on attempt " + attempt);
                }
                return "Success on attempt " + attempt;
            });
        
        // Only retry on RuntimeException
        Function<Throwable, Boolean> retryCondition = 
            throwable -> throwable instanceof RuntimeException;
        
        CompletableFuture<String> result = retryWithBackoff(flakyTask, 3, retryCondition);
        System.out.println("Final: " + result.get());
    }
}

Exception Propagation in Chains

import java.util.concurrent.*;

public class ExceptionPropagation {
    
    public static void main(String[] args) {
        // Exception stops the chain
        CompletableFuture<String> brokenChain = CompletableFuture
            .supplyAsync(() -> {
                throw new RuntimeException("Error in first step");
            })
            .thenApply(s -> {
                System.out.println("This won't execute");
                return s + " transformed";
            })
            .exceptionally(throwable -> {
                System.out.println("Exception caught: " + throwable.getMessage());
                return "Recovered";
            });
        
        System.out.println("Broken chain: " + brokenChain.join());
        
        // Handle exceptions at different points
        CompletableFuture<String> multiStage = CompletableFuture
            .supplyAsync(() -> {
                if (Math.random() > 0.3) {
                    throw new RuntimeException("Stage 1 failed");
                }
                return "Stage 1 success";
            })
            .exceptionally(throwable -> "Stage 1 recovery: " + throwable.getMessage())
            .thenApply(result -> {
                if (Math.random() > 0.3) {
                    throw new RuntimeException("Stage 2 failed");
                }
                return result + " -> Stage 2 success";
            })
            .exceptionally(throwable -> "Stage 2 recovery: " + throwable.getMessage())
            .thenApply(result -> result + " -> Stage 3 success");
        
        System.out.println("Multi-stage: " + multiStage.join());
        
        // Using handle for conditional logic
        CompletableFuture<Integer> parsed = CompletableFuture
            .supplyAsync(() -> "123") // Could be "abc" to cause exception
            .handle((input, throwable) -> {
                if (throwable != null) {
                    return 0; // Default value on error
                }
                try {
                    return Integer.parseInt(input);
                } catch (NumberFormatException e) {
                    return -1; // Parse error
                }
            });
        
        System.out.println("Parsed: " + parsed.join());
    }
}

import java.util.*;

public class ServiceOrchestration {
    
    static class UserService {
        CompletableFuture<String> fetchUser(String userId) {
            return CompletableFuture.supplyAsync(() -> {
                sleep(100);
                return "User-" + userId;
            });
        }
    }
    
    static class OrderService {
        CompletableFuture<List<String>> fetchOrders(String userId) {
            return CompletableFuture.supplyAsync(() -> {
                sleep(150);
                return Arrays.asList("Order-1", "Order-2");
            });
        }
    }
    
    static class InventoryService {
        CompletableFuture<Map<String, Integer>> checkStock(List<String> productIds) {
            return CompletableFuture.supplyAsync(() -> {
                sleep(200);
                Map<String, Integer> stock = new HashMap<>();
                productIds.forEach(id -> stock.put(id, (int)(Math.random() * 10)));
                return stock;
            });
        }
    }
    
    public static CompletableFuture<String> createUserDashboard(String userId) {
        UserService userService = new UserService();
        OrderService orderService = new OrderService();
        InventoryService inventoryService = new InventoryService();
        
        return userService.fetchUser(userId)
            .thenCompose(user -> {
                // Fetch user orders after getting user
                return orderService.fetchOrders(user)
                    .thenCompose(orders -> {
                        // Extract product IDs from orders
                        List<String> productIds = Arrays.asList("prod-1", "prod-2");
                        
                        // Check stock for products
                        return inventoryService.checkStock(productIds)
                            .thenApply(stock -> {
                                // Build dashboard with all data
                                return String.format(
                                    "User: %s\n" +
                                    "Orders: %s\n" +
                                    "Stock: %s",
                                    user, orders, stock
                                );
                            });
                    });
            })
            .exceptionally(throwable -> {
                return "Error creating dashboard: " + throwable.getMessage();
            });
    }
    
    public static void main(String[] args) throws Exception {
        String dashboard = createUserDashboard("123").get();
        System.out.println("User Dashboard:\n" + dashboard);
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

Pattern: Parallel Batch Processing

import java.util.concurrent.*;
import java.util.*;
import java.util.stream.*;

public class BatchProcessing {
    
    public static <T, R> CompletableFuture<List<R>> processInParallel(
            List<T> items,
            Function<T, CompletableFuture<R>> processor,
            int parallelism) {
        
        Executor executor = Executors.newFixedThreadPool(parallelism);
        
        // Create all futures
        List<CompletableFuture<R>> futures = items.stream()
            .map(item -> CompletableFuture.supplyAsync(() -> item, executor)
                .thenCompose(processor))
            .collect(Collectors.toList());
        
        // Combine all results
        return CompletableFuture
            .allOf(futures.toArray(new CompletableFuture[0]))
            .thenApply(v -> futures.stream()
                .map(CompletableFuture::join)
                .collect(Collectors.toList()))
            .whenComplete((result, throwable) -> {
                ((ExecutorService) executor).shutdown();
            });
    }
    
    public static void main(String[] args) throws Exception {
        List<Integer> numbers = IntStream.range(1, 21)
            .boxed()
            .collect(Collectors.toList());
        
        CompletableFuture<List<String>> results = processInParallel(numbers, 
            n -> CompletableFuture.supplyAsync(() -> {
                sleep(100); // Simulate work
                return "Processed-" + n + "-" + Thread.currentThread().getName();
            }), 4); // 4 parallel threads
        
        List<String> processed = results.get();
        System.out.println("Processed " + processed.size() + " items");
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

Pattern: Fan-out Fan-in

import java.util.concurrent.*;
import java.util.*;
import java.util.stream.*;
import java.util.concurrent.Semaphore;

public class FanOutFanIn {
    
    // Fan-out: Process single input to multiple outputs
    // Fan-in: Combine multiple outputs to single result
    
    public static CompletableFuture<Map<String, Object>> processUserData(int userId) {
        // Fan-out: Start multiple async operations
        CompletableFuture<String> userInfo = fetchUserInfo(userId);
        CompletableFuture<List<String>> userOrders = fetchUserOrders(userId);
        CompletableFuture<Double> userBalance = fetchUserBalance(userId);
        CompletableFuture<List<String>> userPreferences = fetchUserPreferences(userId);
        
        // Fan-in: Combine all results
        return CompletableFuture.allOf(userInfo, userOrders, userBalance, userPreferences)
            .thenApply(v -> {
                Map<String, Object> result = new HashMap<>();
                result.put("userInfo", userInfo.join());
                result.put("orders", userOrders.join());
                result.put("balance", userBalance.join());
                result.put("preferences", userPreferences.join());
                return result;
            });
    }
    
    public static CompletableFuture<String> fetchUserInfo(int userId) {
        return CompletableFuture.supplyAsync(() -> {
            sleep(200);
            return "User-" + userId;
        });
    }
    
    public static CompletableFuture<List<String>> fetchUserOrders(int userId) {
        return CompletableFuture.supplyAsync(() -> {
            sleep(300);
            return Arrays.asList("Order-1", "Order-2", "Order-3");
        });
    }
    
    public static CompletableFuture<Double> fetchUserBalance(int userId) {
        return CompletableFuture.supplyAsync(() -> {
            sleep(150);
            return 1250.75;
        });
    }
    
    public static CompletableFuture<List<String>> fetchUserPreferences(int userId) {
        return CompletableFuture.supplyAsync(() -> {
            sleep(250);
            return Arrays.asList("Category-A", "Category-B");
        });
    }
    
    public static void main(String[] args) throws Exception {
        // Single user fan-out/fan-in
        Map<String, Object> userData = processUserData(123).get();
        System.out.println("Single user data: " + userData);
    }
    
    private static void sleep(long ms) {
        try { Thread.sleep(ms); } 
        catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}
