Production Preparation

Before deploying to production, ensure your Spring Boot application is properly configured and optimized.

1. Profile-based Configuration

# application-prod.properties
# Server Configuration
server.port=8080
server.servlet.context-path=/myapp
server.compression.enabled=true
server.tomcat.max-threads=200
server.tomcat.max-connections=10000

# Database Configuration
spring.datasource.url=jdbc:postgresql://prod-db:5432/myapp
spring.datasource.username=${DB_USERNAME}
spring.datasource.password=${DB_PASSWORD}
spring.datasource.hikari.maximum-pool-size=20
spring.datasource.hikari.minimum-idle=5

# JPA Configuration
spring.jpa.hibernate.ddl-auto=validate
spring.jpa.show-sql=false
spring.jpa.properties.hibernate.jdbc.batch_size=50

# Logging Configuration
logging.level.com.myapp=INFO
logging.level.org.springframework=WARN
logging.pattern.console=%d{yyyy-MM-dd HH:mm:ss} - %msg%n
logging.file.name=/var/log/myapp/application.log

# Actuator Configuration
management.endpoints.web.exposure.include=health,info,metrics
management.endpoint.health.show-details=never
management.server.port=8081

# Security
spring.security.oauth2.client.registration.prod.client-id=${OAUTH_CLIENT_ID}
spring.security.oauth2.client.registration.prod.client-secret=${OAUTH_CLIENT_SECRET}
2. External Configuration

# Use environment variables for sensitive data
export DB_USERNAME=prod_user
export DB_PASSWORD=secure_password
export OAUTH_CLIENT_ID=client_id
export OAUTH_CLIENT_SECRET=client_secret

# Or use configuration files
java -jar myapp.jar --spring.config.location=file:/etc/myapp/application.properties
3. Health Checks and Readiness

# Health check configuration
management.endpoint.health.group.readiness.include=db,diskSpace,redis
management.endpoint.health.group.liveness.include=ping
management.endpoint.health.show-details=when-authorized

# For Kubernetes
management.endpoint.health.probes.enabled=true
4. Build Configuration for Production

<!-- Maven pom.xml -->
<build>
    <plugins>
        <plugin>
            <groupId>org.springframework.boot</groupId>

Executable JAR Deployment

Spring Boot's default deployment method - creating standalone executable JAR files.

1. Building Executable JAR

# Maven
mvn clean package -DskipTests

# Gradle
gradle clean build -x test

# The JAR will be created in target/ or build/libs/
# Example: myapp-0.0.1-SNAPSHOT.jar
2. Running the Application

# Basic execution
java -jar myapp-0.0.1-SNAPSHOT.jar

# With production profile
java -jar myapp-0.0.1-SNAPSHOT.jar --spring.profiles.active=prod

# With external configuration
java -jar myapp-0.0.1-SNAPSHOT.jar \
    --spring.config.location=file:/etc/myapp/application.properties

# With JVM options
java -Xms512m -Xmx1024m -jar myapp-0.0.1-SNAPSHOT.jar

# As a Linux service (executable JAR)
chmod +x myapp-0.0.1-SNAPSHOT.jar
./myapp-0.0.1-SNAPSHOT.jar
3. Systemd Service Configuration

# /etc/systemd/system/myapp.service
[Unit]
Description=My Spring Boot Application
After=syslog.target network.target

[Service]
Type=simple
User=appuser
ExecStart=/opt/myapp/myapp-0.0.1-SNAPSHOT.jar
ExecStop=/bin/kill -15 $MAINPID
Restart=on-failure
RestartSec=10

[Install]
WantedBy=multi-user.target
                        

# Enable and start the service
sudo systemctl daemon-reload
sudo systemctl enable myapp.service
sudo systemctl start myapp.service
sudo systemctl status myapp.service
4. JVM Optimization for Production

# Production JVM options
java -server \
     -Xms512m -Xmx1024m \
     -XX:MaxMetaspaceSize=256m \
     -XX:+UseG1GC \
     -XX:MaxGCPauseMillis=200 \
     -XX:ParallelGCThreads=4 \
     -XX:ConcGCThreads=2 \
     -XX:+UnlockExperimentalVMOptions \
     -XX:+UseCGroupMemoryLimitForHeap \
     -Djava.security.egd=file:/dev/./urandom \
     -jar myapp-0.0.1-SNAPSHOT.jar

Docker Deployment

Containerizing Spring Boot applications with Docker for consistent deployments across environments.

1. Basic Dockerfile

# Multi-stage build for optimized image size
FROM eclipse-temurin:11-jdk as builder
WORKDIR /app
COPY .mvn/ .mvn
COPY mvnw pom.xml ./
RUN ./mvnw dependency:go-offline
COPY src ./src
RUN ./mvnw clean package -DskipTests

# Production image
FROM eclipse-temurin:11-jre
RUN addgroup --system --gid 1000 appuser && \
    adduser --system --uid 1000 --ingroup appuser appuser
USER appuser
WORKDIR /app
COPY --from=builder /app/target/*.jar app.jar
EXPOSE 8080
ENTRYPOINT ["java", "-jar", "/app.jar"]
2. Optimized Dockerfile with Layers

FROM eclipse-temurin:11-jre as runtime
RUN addgroup --system --gid 1000 appuser && \
    adduser --system --uid 1000 --ingroup appuser appuser
USER appuser
WORKDIR /app

# Copy the Spring Boot layered JAR
COPY --chown=appuser:appuser target/*.jar app.jar

# Extract layers
RUN java -Djarmode=layertools -jar app.jar extract

# Run with layered JAR for faster startup
ENTRYPOINT ["java", \
    "-Dspring.profiles.active=prod", \
    "-Djava.security.egd=file:/dev/./urandom", \
    "org.springframework.boot.loader.JarLauncher"]
3. Docker Compose for Development

# docker-compose.yml
version: '3.8'
services:
  app:
    build: .
    ports:
      - "8080:8080"
    environment:
      - SPRING_PROFILES_ACTIVE=dev
      - DB_URL=jdbc:postgresql://db:5432/myapp
      - DB_USERNAME=myuser
      - DB_PASSWORD=mypassword
    depends_on:
      - db

      - app-network

  db:
    image: postgres:13
    environment:
      - POSTGRES_DB=myapp
      - POSTGRES_USER=myuser
      - POSTGRES_PASSWORD=mypassword
    volumes:
      - db_data:/var/lib/postgresql/data
    networks:
      - app-network

  redis:
    image: redis:6-alpine
    networks:
      - app-network

volumes:
  db_data:

networks:
  app-network:
    driver: bridge
4. Production Docker Compose

# docker-compose.prod.yml
version: '3.8'
services:
  app:
    image: myregistry/myapp:latest
    restart: unless-stopped

5. Building and Running

# Build the image
docker build -t myapp:latest .

# Run the container
docker run -d \
  --name myapp \
  -p 8080:8080 \
  -e SPRING_PROFILES_ACTIVE=prod \
  -e DB_URL=jdbc:postgresql://host.docker.internal:5432/myapp \
  myapp:latest

# Use Docker Compose
docker-compose up -d

# View logs
docker logs -f myapp

# Stop and remove
docker stop myapp
docker rm myapp

Kubernetes Deployment

Deploying Spring Boot applications to Kubernetes for orchestration, scaling, and high availability.

1. Basic Deployment Manifest

# deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: myapp
  labels:
    app: myapp
spec:
  replicas: 3
  selector:
    matchLabels:
      app: myapp
  template:
    metadata:
      labels:
        app: myapp
    spec:
      containers:
      - name: myapp
        image: myregistry/myapp:latest
        ports:
        - containerPort: 8080
        env:
        - name: SPRING_PROFILES_ACTIVE
          value: "prod"
        - name: DB_URL
          valueFrom:
            secretKeyRef:
              name: db-secret
              key: url
        - name: DB_USERNAME
          valueFrom:
            secretKeyRef:
              name: db-secret
              key: username
        - name: DB_PASSWORD
          valueFrom:
            secretKeyRef:
              name: db-secret
              key: password
        resources:
          requests:
            memory: "256Mi"
            cpu: "250m"
          limits:
            memory: "512Mi"
            cpu: "500m"
        livenessProbe:
          httpGet:
            path: /actuator/health/liveness
            port: 8080
          initialDelaySeconds: 60
          periodSeconds: 10
        readinessProbe:
          httpGet:
            path: /actuator/health/readiness
            port: 8080
          initialDelaySeconds: 30
          periodSeconds: 5

CI/CD Pipelines

Setting up continuous integration and deployment pipelines for Spring Boot applications.

1. GitHub Actions

# .github/workflows/deploy.yml
name: Deploy Spring Boot App

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up JDK 11
      uses: actions/setup-java@v3
      with:
        java-version: '11'
        distribution: 'temurin'
        cache: maven
        
    - name: Build with Maven
      run: mvn clean package -DskipTests
      
    - name: Run tests
      run: mvn test
      
    - name: Build Docker image
      run: docker build -t myapp:${{ github.sha }} .
      
    - name: Run integration tests
      run: docker-compose -f docker-compose.test.yml up --abort-on-container-exit

  deploy:
    needs: build-and-test
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
    
    steps:
    - name: Checkout
      uses: actions/checkout@v3
      
    - name: Build and push Docker image
      run: |
        docker build -t myregistry/myapp:${{ github.sha }} .
        echo ${{ secrets.DOCKER_PASSWORD }} | docker login -u ${{ secrets.DOCKER_USERNAME }} --password-stdin
        docker push myregistry/myapp:${{ github.sha }}
        
    - name: Deploy to Kubernetes
      uses: azure/k8s-deploy@v1
      with:
        namespace: default
        manifests: |
          k8s/deployment.yaml
          k8s/service.yaml
        images: myregistry/myapp:${{ github.sha }}
        kubectl-version: 'latest'
2. Jenkins Pipeline

// Jenkinsfile
pipeline {
    agent any
    
    environment {
        DOCKER_REGISTRY = 'myregistry'
        KUBE_CONFIG = credentials('kubeconfig')
    }
    
    stages {
        stage('Build') {
            steps {
                sh 'mvn clean package -DskipTests'
            }
        }
        
        stage('Test') {
            steps {
                sh 'mvn test'
            }
            post {
                always {
                    junit 'target/surefire-reports/*.xml'
                }
            }
        }
        
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build("${DOCKER_REGISTRY}/myapp:${env.BUILD_ID}")
                }
            }
        }
        
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://myregistry', 'dockerhub') {
                        docker.image("${DOCKER_REGISTRY}/myapp:${env.BUILD_ID}").push()
                    }
                }
            }
        }
        
        stage('Deploy to Staging') {
            steps {
                sh """
                    kubectl set image deployment/myapp \
                    myapp=${DOCKER_REGISTRY}/myapp:${env.BUILD_ID} \
                    --namespace=staging
                """
            }
        }
        
        stage('Integration Tests') {
            steps {
                sh 'mvn verify -P integration-tests'
            }
        }
        
        stage('Deploy to Production') {
            when {
                branch 'main'
            }
            steps {
                input message: 'Deploy to production?'
                sh """
                    kubectl set image deployment/myapp \
                    myapp=${DOCKER_REGISTRY}/myapp:${env.BUILD_ID} \
                    --namespace=production
                """
            }
        }
    }
    
    post {
        always {
            cleanWs()
        }
        success {
            slackSend channel: '#deployments', 
                     message: "Build ${env.BUILD_ID} deployed successfully"
        }
        failure {
            slackSend channel: '#deployments', 
                     message: "Build ${env.BUILD_ID} failed"
        }
    }
}

@Configuration
@EnableJpaRepositories
@EnableTransactionManagement
public class DatabaseConfig {
    
    @Bean
    @ConfigurationProperties("spring.datasource.hikari")
    public DataSource dataSource() {
        return DataSourceBuilder.create().build();
    }
    
    @Bean
    public LocalContainerEntityManagerFactoryBean entityManagerFactory() {
        LocalContainerEntityManagerFactoryBean em = new LocalContainerEntityManagerFactoryBean();
        em.setDataSource(dataSource());
        em.setPackagesToScan("com.myapp.entity");
        
        JpaVendorAdapter vendorAdapter = new HibernateJpaVendorAdapter();
        em.setJpaVendorAdapter(vendorAdapter);
        
        Properties properties = new Properties();
        properties.setProperty("hibernate.hbm2ddl.auto", "validate");
        properties.setProperty("hibernate.dialect", "org.hibernate.dialect.PostgreSQLDialect");
        properties.setProperty("hibernate.show_sql", "false");
        properties.setProperty("hibernate.format_sql", "false");
        properties.setProperty("hibernate.jdbc.batch_size", "50");
        properties.setProperty("hibernate.order_inserts", "true");
        properties.setProperty("hibernate.order_updates", "true");
        properties.setProperty("hibernate.default_batch_fetch_size", "16");
        
        em.setJpaProperties(properties);
        return em;
    }
    
    @Bean
    public PlatformTransactionManager transactionManager(EntityManagerFactory entityManagerFactory) {
        JpaTransactionManager transactionManager = new JpaTransactionManager();
        transactionManager.setEntityManagerFactory(entityManagerFactory);
        return transactionManager;
    }

2. Custom Health Indicators

@Component
public class DatabaseHealthIndicator implements HealthIndicator {
    
    @Autowired
    private DataSource dataSource;
    
    @Override
    public Health health() {
        try (Connection connection = dataSource.getConnection()) {
            boolean isValid = connection.isValid(5);
            if (isValid) {
                return Health.up()
                    .withDetail("database", "Connected")
                    .withDetail("validationQuery", "SUCCESS")
                    .build();
            } else {
                return Health.down()
                    .withDetail("database", "Connection invalid")
                    .build();
            }
        } catch (SQLException e) {
            return Health.down(e)
                .withDetail("database", "Connection failed")
                .build();
        }
    }
}

@Component
public class ExternalServiceHealthIndicator implements HealthIndicator {
    
    @Autowired
    private RestTemplate restTemplate;
    
    @Override
    public Health health() {
        try {
            ResponseEntity response = restTemplate.getForEntity(
                "https://external-service/health", String.class);
                
            if (response.getStatusCode().is2xxSuccessful()) {
                return Health.up()
                    .withDetail("externalService", "Available")
                    .withDetail("responseTime", "OK")
                    .build();
            } else {
                return Health.down()
                    .withDetail("externalService", "Unavailable")
                    .withDetail("statusCode", response.getStatusCodeValue())
                    .build();
            }
        } catch (Exception e) {
            return Health.down(e)
                .withDetail("externalService", "Connection failed")
                .build();
        }
    }
}
3. Custom Metrics

@Service
public class OrderService {
    
    private final MeterRegistry meterRegistry;
    private final Counter orderCounter;
    private final Timer orderProcessingTimer;
    
    public OrderService(MeterRegistry meterRegistry) {
        this.meterRegistry = meterRegistry;
        this.orderCounter = Counter.builder("orders.total")
            .description("Total number of orders")
            .register(meterRegistry);
            
        this.orderProcessingTimer = Timer.builder("orders.processing.time")
            .description("Order processing time")
            .register(meterRegistry);
    }
    
    @Timed(value = "orders.process", description = "Time taken to process order")
    public Order processOrder(Order order) {
        Timer.Sample sample = Timer.start(meterRegistry);
        try {
            // Process order
            Order processed = processOrderBusinessLogic(order);
            orderCounter.increment();
            return processed;
        } finally {
            sample.stop(orderProcessingTimer);
        }
    }
}

@Component
public class CustomMetrics {
    
    private final Gauge activeUsers;
    private final AtomicInteger userCount = new AtomicInteger(0);
    
    public CustomMetrics(MeterRegistry registry) {
        this.activeUsers = Gauge.builder("app.users.active")
            .description("Number of active users")
            .register(registry, userCount);
    }
    
    public void userLoggedIn() {
        userCount.incrementAndGet();
    }
    
    public void userLoggedOut() {
        userCount.decrementAndGet();
    }
}
