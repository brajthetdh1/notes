Docker Installation

Install on Ubuntu


# Update package index
sudo apt-get update

# Install prerequisites
sudo apt-get install \
    apt-transport-https \
    ca-certificates \
    curl \
    gnupg \
    lsb-release

# Add Docker's official GPG key
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /usr/share/keyrings/docker-archive-keyring.gpg

# Add Docker repository
echo \
  "deb [arch=amd64 signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu \
  $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# Install Docker Engine
sudo apt-get update
sudo apt-get install docker-ce docker-ce-cli containerd.io

# Add user to docker group
sudo usermod -aG docker $USER

# Start Docker service
sudo systemctl start docker
sudo systemctl enable docker

Verify Installation


# Check Docker version
docker --version
Docker version 20.10.12, build e91ed57

# Run hello-world container
docker run hello-world
Hello from Docker!
This message shows that your installation appears to be working correctly.

# Check system information
docker system info

Dockerfile for Java Applications

Basic Java Dockerfile


# Use OpenJDK base image
FROM openjdk:11-jre-slim

# Set working directory
WORKDIR /app

# Copy JAR file
COPY target/myapp.jar app.jar

# Expose port
EXPOSE 8080

# Run the application
ENTRYPOINT ["java", "-jar", "app.jar"]

Build and Run


# Build the image
docker build -t my-java-app .

# Run the container
docker run -d -p 8080:8080 --name java-app my-java-app

# Check if it's running
docker ps
CONTAINER ID   IMAGE         COMMAND                  STATUS         PORTS
a1b2c3d4e5f6   my-java-app   "java -jar app.jar"      Up 2 minutes   0.0.0.0:8080->8080/tcp

# Test the application
curl http://localhost:8080

Spring Boot with Docker

Spring Boot Dockerfile


FROM openjdk:11-jre-slim

# Create application user
RUN addgroup --system --gid 1000 appuser && \
    adduser --system --uid 1000 --ingroup appuser appuser

# Set working directory
WORKDIR /app

# Copy JAR file
COPY target/*.jar app.jar

# Change to non-root user
USER appuser

# Expose Spring Boot port
EXPOSE 8080

# Add health check
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
    CMD curl -f http://localhost:8080/actuator/health || exit 1

# JVM options
ENV JAVA_OPTS="-Xmx512m -Xms256m -XX:+UseG1GC -Djava.security.egd=file:/dev/./urandom"

# Run the application
ENTRYPOINT ["sh", "-c", "java $JAVA_OPTS -jar app.jar"]

Maven Build with Docker


# Build the JAR file
mvn clean package -DskipTests

# Build Docker image
docker build -t my-spring-app .

# Run with environment variables
docker run -d \
  --name spring-app \
  -p 8080:8080 \
  -e SPRING_PROFILES_ACTIVE=prod \
  -e DATABASE_URL=jdbc:mysql://host.docker.internal:3306/mydb \
  my-spring-app

# Check health endpoint
curl http://localhost:8080/actuator/health

Multi-stage Builds for Java

Maven Multi-stage Dockerfile


# Build stage
FROM maven:3.8.4-openjdk-11 as builder

WORKDIR /app

# Copy pom.xml for dependency resolution
COPY pom.xml .

# Download dependencies (cached unless pom.xml changes)
RUN mvn dependency:go-offline

# Copy source code
COPY src ./src

# Build application
RUN mvn clean package -DskipTests

# Runtime stage
FROM openjdk:11-jre-slim

# Create non-root user
RUN addgroup --system --gid 1000 appuser && \
    adduser --system --uid 1000 --ingroup appuser appuser

USER appuser
WORKDIR /app

# Copy built JAR from builder stage
COPY --from=builder /app/target/*.jar app.jar

EXPOSE 8080

# Health check
HEALTHCHECK --interval=30s --timeout=3s \
  CMD curl -f http://localhost:8080/actuator/health || exit 1

ENTRYPOINT ["java", "-jar", "/app.jar"]

Gradle Multi-stage Dockerfile


# Build stage
FROM gradle:7.4.2-jdk11 as builder

WORKDIR /app

# Copy build files
COPY build.gradle settings.gradle ./
COPY gradle ./gradle

# Download dependencies
RUN gradle dependencies --no-daemon

# Copy source code
COPY src ./src

# Build application
RUN gradle build --no-daemon -x test

# Runtime stage
FROM openjdk:11-jre-slim

RUN addgroup --system --gid 1000 appuser && \
    adduser --system --uid 1000 --ingroup appuser appuser

USER appuser
WORKDIR /app

COPY --from=builder /app/build/libs/*.jar app.jar

EXPOSE 8080

ENTRYPOINT ["java", "-jar", "/app.jar"]
                            

# Build multi-stage image
docker build -t my-optimized-app .

# Check image size
docker images my-optimized-app
REPOSITORY          TAG       IMAGE ID       CREATED         SIZE
my-optimized-app    latest    a1b2c3d4e5f6   2 minutes ago   235MB

# Compare with single-stage build
docker images my-regular-app
REPOSITORY       TAG       IMAGE ID       CREATED         SIZE
my-regular-app   latest    b2c3d4e5f6a1   5 minutes ago   645MB

Docker Compose for Java Applications

Spring Boot + MySQL Compose


version: '3.8'

services:
  app:
    build: .
    ports:
      - "8080:8080"
    environment:
      - SPRING_PROFILES_ACTIVE=docker
      - SPRING_DATASOURCE_URL=jdbc:mysql://database:3306/myapp
      - SPRING_DATASOURCE_USERNAME=appuser
      - SPRING_DATASOURCE_PASSWORD=apppass
    depends_on:
      - database
    networks:
      - app-network
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8080/actuator/health"]
      interval: 30s
      timeout: 10s
      retries: 3

  database:
    image: mysql:8.0
    environment:
      MYSQL_ROOT_PASSWORD: rootpass
      MYSQL_DATABASE: myapp
      MYSQL_USER: appuser
      MYSQL_PASSWORD: apppass
    volumes:
      - db_data:/var/lib/mysql
    networks:
      - app-network
    ports:
      - "3306:3306"

volumes:
  db_data:

networks:
  app-network:
    driver: bridge

# Run multiple containers on same network
docker run -d --name database --network app-network mysql:8.0

docker exec app ping database

Docker Volumes for Data Persistence

Volume Commands


# List volumes
docker volume ls

# Create a volume
docker volume create app-data

# Inspect volume
docker volume inspect app-data

# Run container with volume
docker run -d \
  --name app \
  -v app-data:/app/data \
  my-java-app

# Use bind mount for development
docker run -d \
  --name app \
  -v $(pwd)/data:/app/data \
  -v $(pwd)/logs:/app/logs \
  my-java-app

# Remove unused volumes
docker volume prune

Production Deployment

Production Dockerfile


FROM eclipse-temurin:11-jre-alpine

# Security: Create non-root user
RUN addgroup -S appgroup && adduser -S appuser -G appgroup

# Install curl for health checks
RUN apk add --no-cache curl

USER appuser
WORKDIR /app

COPY --chown=appuser:appgroup target/*.jar app.jar

EXPOSE 8080

# JVM production settings
ENV JAVA_OPTS="-Xmx512m -Xms256m \
-XX:+UseG1GC \
-XX:MaxGCPauseMillis=200 \
-XX:+UnlockExperimentalVMOptions \
-XX:+UseCGroupMemoryLimitForHeap \
-Djava.security.egd=file:/dev/./urandom \
-Dspring.profiles.active=prod"

# Health check
HEALTHCHECK --interval=30s --timeout=3s --start-period=60s --retries=3 \
    CMD curl -f http://localhost:8080/actuator/health || exit 1

ENTRYPOINT ["sh", "-c", "java $JAVA_OPTS -jar app.jar"]

Production Deployment Commands


# Build for production
docker build -t myapp:prod .

# Run with resource limits
docker run -d \
  --name myapp-prod \
  -p 8080:8080 \
  --memory=512m \
  --cpus=1.0 \
  --restart=unless-stopped \
  -e SPRING_PROFILES_ACTIVE=prod \
  myapp:prod

# Check resource usage
docker stats myapp-prod

# Set up logging driver
docker run -d \
  --name myapp-prod \
  --log-driver=json-file \
  --log-opt max-size=10m \
  --log-opt max-file=3 \
  myapp:prod

# Update application (blue-green deployment)
# 1. Build new version
docker build -t myapp:2.0 .

# 2. Run new version
docker run -d --name myapp-v2 -p 8081:8080 myapp:2.0

# 3. Test new version
curl http://localhost:8081/actuator/health

# 4. Switch traffic (update load balancer)

# 5. Stop old version
docker stop myapp-prod

Troubleshooting Common Issues

Debugging Commands


# View container logs
docker logs myapp

# Follow logs in real-time
docker logs -f myapp

# Execute into running container
docker exec -it myapp bash

# View container details
docker inspect myapp

# Check container processes
docker top myapp

# Monitor resource usage
docker stats myapp

# View system events
docker system events

Common Java Issues


# Container exits immediately
docker logs myapp
Error: Unable to access jarfile app.jar

# Fix: Check JAR file path in Dockerfile

# Out of memory error
docker logs myapp
java.lang.OutOfMemoryError: Java heap space

# Fix: Increase memory limits
docker run -d --memory=1g myapp

# Connection refused to database
# Fix: Use container names in connection strings
# jdbc:mysql://database:3306/mydb (not localhost)

# File permission issues
# Fix: Use non-root user and proper file ownership

Useful Debugging Tips


# Build with debug output
docker build --progress=plain -t myapp .

# Run with interactive shell for debugging
docker run -it --entrypoint /bin/bash myapp

# Check if port is accessible
docker exec myapp curl http://localhost:8080/actuator/health

# Copy files from container for inspection
docker cp myapp:/app/logs/application.log ./debug.log

# Check DNS resolution inside container
docker exec myapp nslookup database

# Monitor container resource usage
docker stats myapp
